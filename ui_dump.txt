/// The complete visitor feedback flow — PROJECT_PROCESS.md §10.6, I24.
///
/// Watches the `FeedbackController` state machine and renders one view per
/// state: picker (idle/selecting/submitting) → thank-you (success) →
/// rating prompt (after a short beat) → cooldown/error slots. The prompt's
/// Yes navigates to `/rating` (a placeholder until I29); Skip resets the
/// controller back to idle.
class FeedbackScreen extends ConsumerStatefulWidget {
  const FeedbackScreen({
    super.key,
    this.thankYouDelay = const Duration(seconds: 2),
  });

  /// How long the thank-you is held before the rating prompt appears.
  /// Injectable so the E2E test advances it instantly.
  final Duration thankYouDelay;

  @override
  ConsumerState<FeedbackScreen> createState() => _FeedbackScreenState();
}

class _FeedbackScreenState extends ConsumerState<FeedbackScreen> {
  bool _showPrompt = false;
  Timer? _promptTimer;

  @override
  void dispose() {
    _promptTimer?.cancel();
    super.dispose();
  }

  void _onState(FeedbackState next) {
    if (next is FeedbackSuccess) {
      _promptTimer?.cancel();
      _promptTimer = Timer(widget.thankYouDelay, () {
        if (mounted) {
          setState(() => _showPrompt = true);
        }
      });
      return;
    }

    // Leaving success cancels the beat: no stale prompt on the next visit.
    _promptTimer?.cancel();
    _promptTimer = null;
    if (_showPrompt) {
      setState(() => _showPrompt = false);
    }
  }

  Future<void> _select(EmojiChoice choice) =>
      ref.read(feedbackControllerProvider.notifier).select(choice);

  void _reset() => ref.read(feedbackControllerProvider.notifier).reset();

  @override
  Widget build(BuildContext context) {
    // Riverpod 3 requires listeners to be registered during build.
    ref.listen(feedbackControllerProvider, (previous, next) => _onState(next));
    final state = ref.watch(feedbackControllerProvider);
    return AttractOverlay(child: KioskScaffold(child: _body(state)));
  }

  Widget _body(FeedbackState state) {
    if (_showPrompt && state is FeedbackSuccess) {
      return Center(
        child: RateProjectsPrompt(
          onYes: () => context.go('/rating'),
          onSkip: _reset,
        ),
      );
    }

    return switch (state) {
      FeedbackIdle() || FeedbackSelecting() || FeedbackSubmitting() => Center(
          child: SingleChildScrollView(
            padding: const EdgeInsets.all(AppSpacing.xl),
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                Text(
                  'How was your visit?',
                  textAlign: TextAlign.center,
                  style: AppTypography.headline.copyWith(
                    color: AppColors.inkPrimary,
                  ),
                ),
                const SizedBox(height: AppSpacing.xxl),
                EmojiPicker(
                  onSelected: (choice) => unawaited(_select(choice)),
                  autofocusFirst: true,
                ),
              ],
            ),
          ),
        ),
      FeedbackSuccess() => const Center(child: ThankYouCard()),
      FeedbackCooldown() => CooldownScreen(
          remaining: ref.watch(feedbackCooldownRemainingProvider),
          onExpiry: _reset,
          message: state.message,
        ),
      FeedbackError() => Center(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              Text(
                state.message,
                textAlign: TextAlign.center,
                style: AppTypography.body.copyWith(color: AppColors.inkPrimary),
              ),
              const SizedBox(height: AppSpacing.xl),
              FilledButton(onPressed: _reset, child: const Text('Try again')),
            ],
          ),
        ),
    };
  }
}

/// Full rating flow — carousel → project → stars → submit → confirmation
/// (§10.7, I29). Local pick/score stay here; persistence goes through the
/// tested `RatingController` → `SubmitRating` → local-first queue.
class RatingScreen extends ConsumerStatefulWidget {
  const RatingScreen({super.key});

  static const submitKey = Key('rating_submit');
  static const confirmationKey = Key('rating_confirmation');
  static const statusKey = Key('rating_status');

  @override
  ConsumerState<RatingScreen> createState() => _RatingScreenState();
}

class _RatingScreenState extends ConsumerState<RatingScreen> {
  Project? _selected;
  int _score = 0;

  void _select(Project project) {
    setState(() {
      _selected = project;
      _score = 0;
    });
  }

  void _scoreChanged(int score) => setState(() => _score = score);

  void _submit() {
    final project = _selected;
    if (project == null || _score == 0) {
      return;
    }
    unawaited(
      ref.read(ratingControllerProvider.notifier).submit(
            projectId: project.id,
            score: _score,
          ),
    );
  }

  void _done() {
    ref.read(ratingControllerProvider.notifier).reset();
    setState(() {
      _selected = null;
      _score = 0;
    });
  }

  @override
  Widget build(BuildContext context) {
    final state = ref.watch(ratingControllerProvider);

    return AttractOverlay(
      child: KioskScaffold(
        child: Column(
        children: [
          Align(
            alignment: Alignment.topLeft,
            child: TextButton(
              onPressed: () => context.go('/feedback'),
              child: const Text('← Back'),
            ),
          ),
          Expanded(child: Center(child: _body(state))),
        ],
      ),
      ),
    );
  }

  Widget _body(RatingState state) {
    return switch (state) {
      RatingSuccess() => _confirmation(),
      RatingSubmitting() => _status('Saving your rating…'),
      RatingCooldown() => KeyedSubtree(
          key: RatingScreen.statusKey,
          child: CooldownScreen(
            remaining: ref.watch(ratingCooldownRemainingProvider),
            onExpiry: _done,
            message: state.message,
          ),
        ),
      RatingError() => KeyedSubtree(
          key: RatingScreen.statusKey,
          child: _statusCopy(state.message),
        ),
      RatingIdle() => _selected == null ? _carousel() : _stars(),
    };
  }

  Widget _carousel() {
    return SingleChildScrollView(
      padding: const EdgeInsets.all(AppSpacing.xl),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          Text(
            'Rate our projects?',
            textAlign: TextAlign.center,
            style: AppTypography.headline.copyWith(
              color: AppColors.inkPrimary,
            ),
          ),
          const SizedBox(height: AppSpacing.sm),
          Text(
            'Swipe or use ← →, then tap a project.',
            textAlign: TextAlign.center,
            style: AppTypography.caption.copyWith(
              color: AppColors.inkSecondary,
            ),
          ),
          const SizedBox(height: AppSpacing.xl),
          ProjectCarousel(
            projects: ref.watch(projectFixturesProvider),
            onSelect: _select,
          ),
        ],
      ),
    );
  }

  Widget _stars() {

    return SingleChildScrollView(
      padding: const EdgeInsets.all(AppSpacing.xl),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          Text(
            project.title,
            textAlign: TextAlign.center,
            style: AppTypography.headline.copyWith(
              color: AppColors.inkPrimary,
            ),
          ),
          const SizedBox(height: AppSpacing.sm),
          Text(
            'How would you rate it?',
            textAlign: TextAlign.center,
            style: AppTypography.body.copyWith(
              color: AppColors.inkSecondary,
            ),
          ),
          const SizedBox(height: AppSpacing.lg),
          IgnorePointer(
            child: ProjectCard(project: project, onSelect: (_) {}),
          ),
          const SizedBox(height: AppSpacing.lg),
          RatingWidget(onChanged: _scoreChanged, autofocusFirst: true),
          const SizedBox(height: AppSpacing.xl),
          KeyedSubtree(
            key: RatingScreen.submitKey,
            child: FilledButton(

/// Composes the Living Sketchbook [ThemeData] from design tokens.
/// PROJECT_PROCESS.md §4.1 (`core/theme/app_theme.dart`) and §5.
///
/// Two deliberate API surfaces:
///
/// * [light] is **pure** — tokens in, [ThemeData] out. No binding, no
///   network, fully hermetic for unit/widget tests (§8.1).
/// * [loadFonts] registers the Patrick Hand face with the engine. It is
///   bootstrap-only (called from `main()`), never from tests.
abstract final class AppTheme {
  /// The single source of truth for app theming (no hardcoded values).
  static ThemeData get light => ThemeData(
        useMaterial3: true,
        colorScheme: const ColorScheme.light(
          primary: AppColors.accentWarm,
          onPrimary: AppColors.paperWhite,
          secondary: AppColors.accentSky,
          onSecondary: AppColors.inkPrimary,
          surface: AppColors.paperWhite,
          onSurface: AppColors.inkPrimary,
          error: AppColors.accentBlush,
          onError: AppColors.inkPrimary,
        ),
        scaffoldBackgroundColor: AppColors.paperBackground,
        textTheme: const TextTheme(
          displayLarge: AppTypography.display,
          headlineMedium: AppTypography.headline,
          bodyLarge: AppTypography.body,
          bodyMedium: AppTypography.body,
          bodySmall: AppTypography.caption,
        ),
        cardTheme: CardThemeData(
          color: AppColors.paperWhite,
          elevation: 0,
          margin: const EdgeInsets.symmetric(
            horizontal: AppSpacing.sm,
            vertical: AppSpacing.sm,
          ),
          shape: RoundedRectangleBorder(
            borderRadius: BorderRadius.circular(AppSpacing.md),
          ),
        ),
        // Focus is a first-class state for D-pad (§2.3.2): warm halo built
        // from the accent token.
        focusColor: AppColors.accentWarm.withValues(alpha: 0.25),
        // Ripple instead of M3's sparkle: cheaper on low-end GPUs (§2.3.3).
        splashFactory: InkRipple.splashFactory,
      );

  /// Registers the Patrick Hand face (google_fonts). Call once from
  /// `main()` after `WidgetsFlutterBinding.ensureInitialized()`.
  ///
  /// Deferred to bootstrap so [light] stays hermetic and tests never
  /// trigger network or binding access.
  static void loadFonts() {
    GoogleFonts.patrickHand();
  }
}

/// Living Sketchbook typography tokens — PROJECT_PROCESS.md §5.3.
///
/// Family locked at Iteration 4: **Patrick Hand** — a handwriting face
/// designed for readability, which matters at 10-foot TV distance; the §13

abstract final class AppTypography {
  /// Registered font family name (must match google_fonts' registration).
  static const String family = 'PatrickHand';

  // --- Size tokens: §5.3 ranges, concrete values locked here. ---

  /// TV headlines — §5.3 range 48–64sp, locked at 56.
  static const double displaySize = 56;

  /// Section headlines — §5.3 range 32–40sp, locked at 36.
  static const double headlineSize = 36;

  /// Body — §5.3 range 20–24sp, locked at 22 (phone-first; the TV audit at
  /// Iteration 42 promotes anything below [tvMinimumSize]).
  static const double bodySize = 22;

  /// Captions — §5.3 hard minimum.
  static const double captionSize = 16;

  /// §5.5 TV floor: anything a visitor *must* read on TV is ≥ 32sp.
  static const double tvMinimumSize = 32;

  // --- Styles (sizes only; colors are decided by the theme, §2.1.4). ---

  static const TextStyle display =
      TextStyle(fontFamily: family, fontSize: displaySize);

  static const TextStyle headline =
      TextStyle(fontFamily: family, fontSize: headlineSize);

  static const TextStyle body =
      TextStyle(fontFamily: family, fontSize: bodySize);

  static const TextStyle caption =
      TextStyle(fontFamily: family, fontSize: captionSize);

/// Kiosk attract mode: after [idleDuration] without input, a pencil-drawn
/// welcome loop covers the screen; any touch or key dismisses it and resets
/// the timer (§5.1, §13).
///
/// The wrapper deliberately does **not** autofocus — key events bubble up
/// from whatever the screen focused (first face, first star), so initial
/// focus targets stay intact. The visual is `IgnorePointer`; input is caught
/// by the wrapper's [Listener] and [Focus] instead.
class AttractOverlay extends StatefulWidget {
  const AttractOverlay({
    super.key,
    required this.child,
    this.idleDuration = const Duration(seconds: 20),
  });

  final Widget child;

  /// Tunable per §13; defaults to the 20s attract loop.
  final Duration idleDuration;

  static const overlayKey = Key('attract_overlay');

  @override
  State<AttractOverlay> createState() => _AttractOverlayState();
}

class _AttractOverlayState extends State<AttractOverlay>
    with SingleTickerProviderStateMixin {
  Timer? _timer;
  bool _idle = false;

  /// Loops the "pencil drawing" underline; no asset, no network.
  late final AnimationController _loop = AnimationController(
    vsync: this,
    duration: const Duration(seconds: 2),
  )..repeat();

  @override
  void initState() {
    super.initState();
    _arm();
  }

  @override
  void dispose() {
    _timer?.cancel();
    _loop.dispose();
    super.dispose();
  }

  void _arm() {
    _timer?.cancel();
    _timer = Timer(widget.idleDuration, _goIdle);
  }

  void _wake() {
    _arm();
    if (_idle && mounted) {
      setState(() => _idle = false);
    }
  }

  void _goIdle() {
    if (mounted) {
      setState(() => _idle = true);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Listener(
      onPointerDown: (_) => _wake(),
      child: Focus(
        onKeyEvent: (node, event) {
          _wake();
          // Let the screen keep handling its own keys.
          return KeyEventResult.ignored;
        },
        child: Stack(
          children: [
            widget.child,
            if (_idle)
              Positioned.fill(
                child: IgnorePointer(
                  child: KeyedSubtree(
                    key: AttractOverlay.overlayKey,
                    child: ColoredBox(
                      color: AppColors.paperBackground.withValues(alpha: 0.96),
                      child: Center(
                        child: Column(
                          mainAxisSize: MainAxisSize.min,
                          children: [
                            Text(
                              'Welcome to UCSTGO',
                              textAlign: TextAlign.center,
                              style: AppTypography.display.copyWith(
                                color: AppColors.inkPrimary,
                              ),
                            ),
                            const SizedBox(height: AppSpacing.md),
                            SizedBox(
                              width: 240,
                              height: 16,
                              child: AnimatedBuilder(
                                animation: _loop,
                                builder: (context, _) => CustomPaint(
                                  painter: _PencilUnderlinePainter(
                                    progress: _loop.value,
                                    color: AppColors.accentWarm,
                                  ),
                                ),
                              ),
                            ),
                            const SizedBox(height: AppSpacing.md),
                            Text(
                              'Touch or press any key to begin',
                              textAlign: TextAlign.center,
                              style: AppTypography.caption.copyWith(
                                color: AppColors.inkSecondary,
                              ),
                            ),
                          ],
                        ),
                      ),
                    ),
                  ),
                ),
              ),
          ],
        ),
      ),
    );
  }
}

/// A wobbly line that draws itself from left to right, looping forever.
class _PencilUnderlinePainter extends CustomPainter {
  const _PencilUnderlinePainter({
    required this.progress,
    required this.color,
  });

  final double progress;
  final Color color;

  @override
  void paint(Canvas canvas, Size size) {
    if (progress <= 0) {
      return;
    }

    final end = size.width * progress;
    final path = Path();
    const steps = 24;
    for (var step = 0; step <= steps; step++) {
      final x = end * step / steps;
      final y = size.height / 2 + (step.isEven ? 1.5 : -1.5);
      if (step == 0) {
        path.moveTo(x, y);
      } else {
        path.lineTo(x, y);
      }
    }

    canvas.drawPath(
      path,
      Paint()
        ..color = color
        ..style = PaintingStyle.stroke
        ..strokeWidth = 2.5
        ..strokeCap = StrokeCap.round,
    );
  }

  @override
  bool shouldRepaint(covariant _PencilUnderlinePainter oldDelegate) =>
      oldDelegate.progress != progress || oldDelegate.color != color;
}

/// (§5.1, §7): a doodled clock counts down the persisted deadline.
///
/// Driven by an injected [Stream] so tests never touch platform storage;
/// when the stream reaches zero (or completes) [onExpiry] fires exactly once

class CooldownScreen extends StatefulWidget {
  const CooldownScreen({

    required this.remaining,
    this.onExpiry,
    this.message = 'Thanks! Take a breath before going again.',
  });

  /// Remaining time from `CooldownService.remaining(action)`.
  final Stream<Duration> remaining;

  /// Called once when the cooldown ends.
  final VoidCallback? onExpiry;

  final String message;

  static const clockKey = Key('cooldown_clock');
  static const countdownKey = Key('cooldown_countdown');

  @override
  State<CooldownScreen> createState() => _CooldownScreenState();
}

class _CooldownScreenState extends State<CooldownScreen> {
  StreamSubscription<Duration>? _subscription;
  Duration? _remaining;
  bool _expired = false;

  @override
  void initState() {
    super.initState();
    _listen(widget.remaining);
  }

  @override
  void didUpdateWidget(covariant CooldownScreen oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (oldWidget.remaining != widget.remaining) {
      _subscription?.cancel();
      _listen(widget.remaining);
    }
  }

  @override
  void dispose() {
    _subscription?.cancel();
    super.dispose();
  }

  void _listen(Stream<Duration> stream) {
    _remaining = null;
    _expired = false;
    _subscription = stream.listen(
      (left) {
        if (!mounted) {
          return;
        }
        setState(() => _remaining = left);
        if (left <= Duration.zero) {
          _fireExpiry();
        }
      },
      onDone: _fireExpiry,
      onError: (_) => _fireExpiry(),
    );
  }

  void _fireExpiry() {
    if (_expired) {
      return;
    }
    _expired = true;
    widget.onExpiry?.call();
  }

  @override
  Widget build(BuildContext context) {
    final left = _remaining;
    final seconds = left?.inSeconds;
    final fraction = seconds == null || seconds <= 0
        ? 0.0
        : (seconds / 30).clamp(0.0, 1.0);

    return Center(
      child: SingleChildScrollView(
        padding: const EdgeInsets.all(AppSpacing.lg),
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
          SizedBox.square(
            dimension: 160,

              key: CooldownScreen.clockKey,

            style: AppTypography.display.copyWith(

    );
  }
}

/// Wobbly clock face with a hand that unwinds as the deadline approaches.
class _ClockPainter extends CustomPainter {
  const _ClockPainter({
    required this.ink,
    required this.glow,
    required this.fraction,
  });

  final Color ink;
  final Color glow;
  final double fraction;

  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height / 2);
    final radius = size.shortestSide * 0.42;

    canvas.drawCircle(
      center,
      radius,
      Paint()..color = glow.withValues(alpha: 0.25),
    );

    final stroke = Paint()
      ..color = ink
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2.6
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;

    canvas.drawCircle(center, radius, stroke);

    for (var tick = 0; tick < 12; tick++) {
      final angle = tick * 2 * math.pi / 12;
      final inner = radius * 0.82;
      final outer = radius * 0.92;
      canvas.drawLine(
        center + Offset(math.cos(angle) * inner, math.sin(angle) * inner),
        center + Offset(math.cos(angle) * outer, math.sin(angle) * outer),
        stroke..strokeWidth = 1.6,
      );
    }

    final handAngle = -math.pi / 2 + (1 - fraction) * 2 * math.pi;
    canvas.drawLine(
      center,
      center +
          Offset(math.cos(handAngle) * radius * 0.6,
              math.sin(handAngle) * radius * 0.6),
      stroke
        ..strokeWidth = 3
        ..color = ink,
    );
    canvas.drawCircle(center, 4, Paint()..color = ink);
  }

  @override
  bool shouldRepaint(covariant _ClockPainter oldDelegate) =>
      oldDelegate.fraction != fraction ||
      oldDelegate.ink != ink ||
      oldDelegate.glow != glow;
}

/// Paints one face as three or four wobbly ink strokes on paper.
///
/// Deliberately cheap: a handful of paths per face, one `CustomPaint` each,
/// no shaders or save layers — safe for a low-end TV GPU (§2.3.3).
class _EmojiFacePainter extends CustomPainter {
  const _EmojiFacePainter({
    required this.choice,
    required this.ink,
    required this.accent,
    required this.blush,
  });

  final EmojiChoice choice;
  final Color ink;
  final Color accent;
  final Color blush;

  static const double _washOpacity = 0.16;
  static const double _blushOpacity = 0.5;

  /// Deterministic wobble phase per face so repaints never shimmer.
  int get _seed => switch (choice) {
        EmojiChoice.satisfied => 3,
        EmojiChoice.neutral => 11,
        EmojiChoice.dissatisfied => 19,
      };

  @override
  void paint(Canvas canvas, Size size) {
    final side = size.shortestSide;
    final center = Offset(size.width / 2, size.height / 2);

    // A soft accent swipe behind the face, like a marker highlight.
    final wash = Paint()
      ..color = accent.withValues(alpha: _washOpacity)
      ..style = PaintingStyle.fill;
    canvas.drawPath(
      _blob(
        center.translate(side * 0.01, side * 0.015),
        side * 0.46,
        seed: _seed + 5,
      ),
      wash,
    );

    final stroke = Paint()
      ..color = ink
      ..style = PaintingStyle.stroke
      ..strokeWidth = side * 0.018
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;

    // A faint second pass, offset by a hair — the "drawn twice" pencil look.
    final ghost = Paint()
      ..color = ink.withValues(alpha: 0.28)
      ..style = PaintingStyle.stroke
      ..strokeWidth = side * 0.012
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;
    canvas.drawPath(
      _blob(
        center.translate(-side * 0.007, side * 0.006),
        side * 0.405,
        seed: _seed + 1,
      ),
      ghost,
    );

    canvas.drawPath(_blob(center, side * 0.41, seed: _seed), stroke);
    _paintFeatures(canvas, side, center, stroke);
  }

  void _paintFeatures(Canvas canvas, double side, Offset center, Paint stroke) {
    final eyeY = center.dy - side * 0.07;
    final leftEye = Offset(center.dx - side * 0.16, eyeY);
    final rightEye = Offset(center.dx + side * 0.16, eyeY);

    if (choice == EmojiChoice.satisfied) {
      _drawClosedEye(canvas, leftEye, side, stroke);
      _drawClosedEye(canvas, rightEye, side, stroke);
      _drawCheeks(canvas, center, side);
    } else {
      _drawDotEye(canvas, leftEye, side);
      _drawDotEye(canvas, rightEye, side);
    }

    _drawMouth(canvas, center, side, stroke);
  }

  /// A happy "^" eye: an upward-bowed arc.
  void _drawClosedEye(Canvas canvas, Offset eye, double side, Paint stroke) {
    final width = side * 0.13;
    final path = Path()
      ..moveTo(eye.dx - width / 2, eye.dy + width * 0.12)
      ..quadraticBezierTo(
        eye.dx,
        eye.dy - width * 0.45,
        eye.dx + width / 2,
        eye.dy + width * 0.12,
      );
    canvas.drawPath(path, stroke);
  }

  void _drawDotEye(Canvas canvas, Offset eye, double side) {
    final dot = Paint()
      ..color = ink
      ..style = PaintingStyle.fill;
    canvas.drawCircle(eye, side * 0.027, dot);
  }

  void _drawCheeks(Canvas canvas, Offset center, double side) {
    final cheek = Paint()
      ..color = blush.withValues(alpha: _blushOpacity)
      ..style = PaintingStyle.fill;
    final y = center.dy + side * 0.1;
    canvas.drawCircle(
      Offset(center.dx - side * 0.27, y),
      side * 0.05,
      cheek,
    );
    canvas.drawCircle(
      Offset(center.dx + side * 0.27, y),
      side * 0.05,
      cheek,
    );
  }

  void _drawMouth(Canvas canvas, Offset center, double side, Paint stroke) {
    final halfWidth = side * 0.15;
    final mouthY = center.dy + side * 0.12;
    final path = Path();

    switch (choice) {
      case EmojiChoice.satisfied:
        // A big "U" smile.
        path
          ..moveTo(center.dx - halfWidth, mouthY)
          ..quadraticBezierTo(
            center.dx,
            mouthY + side * 0.16,
            center.dx + halfWidth,
            mouthY,
          );
      case EmojiChoice.neutral:
        // A small wavy line — not a frown, not a smile.
        path
          ..moveTo(center.dx - halfWidth, mouthY + side * 0.02)
          ..quadraticBezierTo(
            center.dx - halfWidth / 2,
            mouthY - side * 0.01,
            center.dx,
            mouthY + side * 0.02,
          )
          ..quadraticBezierTo(
            center.dx + halfWidth / 2,
            mouthY + side * 0.05,
            center.dx + halfWidth,
            mouthY + side * 0.02,
          );
      case EmojiChoice.dissatisfied:
        // An inverted "U" frown: corners down, centre up.
        path
          ..moveTo(center.dx - halfWidth, mouthY + side * 0.07)
          ..quadraticBezierTo(
            center.dx,
            mouthY - side * 0.1,
            center.dx + halfWidth,
            mouthY + side * 0.07,
          );
    }

    canvas.drawPath(path, stroke);
  }

  /// A closed but slightly irregular circle, sampled with two low-frequency
  /// sine waves — hand-drawn without randomness.
  Path _blob(Offset center, double radius, {required int seed}) {
    const steps = 48;
    final path = Path();

    for (var i = 0; i <= steps; i++) {
      final angle = 2 * math.pi * i / steps;
      final wobble = 1 +
          0.018 * math.sin(3 * angle + seed) +
          0.011 * math.sin(5 * angle - seed * 1.7);
      final point = center +
          Offset(
            math.cos(angle) * radius * wobble,
            math.sin(angle) * radius * wobble,
          );

      if (i == 0) {
        path.moveTo(point.dx, point.dy);
      } else {
        path.lineTo(point.dx, point.dy);
      }
    }

    return path..close();
  }

  @override
  bool shouldRepaint(covariant _EmojiFacePainter oldDelegate) =>
      oldDelegate.choice != choice ||
      oldDelegate.ink != ink ||
      oldDelegate.accent != accent ||
      oldDelegate.blush != blush;
}

/// Faint dot-grid "paper texture" (§5.1).
///

class _DotGridPainter extends CustomPainter {
  const _DotGridPainter();

  static const double _dotOpacity = 0.06;
  static const double _dotRadius = 1.2;

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = AppColors.inkSecondary.withValues(alpha: _dotOpacity)
      ..strokeWidth = _dotRadius * 2
      ..strokeCap = StrokeCap.round;

    final points = <Offset>[
      for (var x = AppSpacing.xl / 2; x < size.width; x += AppSpacing.xl)
        for (var y = AppSpacing.xl / 2; y < size.height; y += AppSpacing.xl)
          Offset(x, y),
    ];

    canvas.drawPoints(PointMode.points, points, paint);
  }

  @override
  bool shouldRepaint(covariant _DotGridPainter oldDelegate) => false;

/// Wobbly strip of masking tape holding the polaroid up.
class _TapePainter extends CustomPainter {
  const _TapePainter({required this.tape, required this.edge});

  final Color tape;
  final Color edge;

  @override
  void paint(Canvas canvas, Size size) {
    final strip = Path()
      ..moveTo(3, 4)
      ..lineTo(size.width - 4, 1)
      ..lineTo(size.width - 1, size.height - 3)
      ..lineTo(2, size.height - 5)
      ..close();

    canvas.drawPath(strip, Paint()..color = tape..style = PaintingStyle.fill);
    canvas.drawPath(
      strip,
      Paint()
        ..color = edge.withValues(alpha: 0.3)
        ..style = PaintingStyle.stroke
        ..strokeWidth = 1.2
        ..strokeJoin = StrokeJoin.round,
    );
  }

  @override
  bool shouldRepaint(covariant _TapePainter oldDelegate) =>
      oldDelegate.tape != tape || oldDelegate.edge != edge;
}

/// A cheap doodle scene standing in for a real project image.
class _PlaceholderArtPainter extends CustomPainter {
  const _PlaceholderArtPainter({
    required this.ink,
    required this.sky,
    required this.warm,
  });

  final Color ink;
  final Color sky;
  final Color warm;

  @override
  void paint(Canvas canvas, Size size) {
    final w = size.width;
    final h = size.height;

    canvas.drawCircle(
      Offset(w * 0.76, h * 0.24),
      w * 0.1,
      Paint()..color = warm.withValues(alpha: 0.5),
    );

    final stroke = Paint()
      ..color = ink
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2.4
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;

    canvas.drawPath(
      Path()
        ..moveTo(w * 0.06, h * 0.8)
        ..lineTo(w * 0.4, h * 0.34)
        ..lineTo(w * 0.7, h * 0.8),
      stroke,
    );
    canvas.drawPath(
      Path()
        ..moveTo(w * 0.38, h * 0.8)
        ..lineTo(w * 0.66, h * 0.5)
        ..lineTo(w * 0.94, h * 0.8),
      stroke..color = sky,
    );
    canvas.drawLine(
      Offset(w * 0.06, h * 0.84),
      Offset(w * 0.94, h * 0.82),
      stroke..color = ink,
    );
  }

  @override
  bool shouldRepaint(covariant _PlaceholderArtPainter oldDelegate) =>
      oldDelegate.ink != ink ||
      oldDelegate.sky != sky ||

/// Hand-drawn page dots: wobbly ellipses, the active one filled with the
/// warm accent. No animation controller — cheap on a low-end TV (§2.3.3).
class _PageDotsPainter extends CustomPainter {
  const _PageDotsPainter({
    required this.count,
    required this.active,
    required this.ink,
    required this.accent,
  });

  final int count;
  final int active;
  final Color ink;
  final Color accent;

  @override
  void paint(Canvas canvas, Size size) {
    final spacing = size.width / (count + 1);

    for (var i = 0; i < count; i++) {
      final isActive = i == active;
      final center = Offset(
        spacing * (i + 1),
        size.height / 2 + (i.isEven ? 1 : -1),
      );
      final radius = isActive ? 6.0 : 4.0;
      final path = Path();
      const steps = 16;

      for (var step = 0; step <= steps; step++) {
        final angle = 2 * math.pi * step / steps;
        final wobble = 1 + 0.08 * math.sin(3 * angle + i);
        final point = center +
            Offset(
              math.cos(angle) * radius * wobble,
              math.sin(angle) * radius * wobble,
            );
        if (step == 0) {
          path.moveTo(point.dx, point.dy);
        } else {
          path.lineTo(point.dx, point.dy);
        }
      }

      canvas.drawPath(
        path..close(),
        Paint()
          ..color = isActive ? accent : ink.withValues(alpha: 0.4)
          ..style = isActive ? PaintingStyle.fill : PaintingStyle.stroke
          ..strokeWidth = 2,
      );
    }
  }

  @override
  bool shouldRepaint(covariant _PageDotsPainter oldDelegate) =>
      oldDelegate.count != count ||
      oldDelegate.active != active ||
      oldDelegate.ink != ink ||
      oldDelegate.accent != accent;
}

/// Draws a wobbly five-point star with a faint second pencil pass.
class _StarPainter extends CustomPainter {
  const _StarPainter({
    required this.progress,
    required this.ink,
    required this.gold,
  });

  final double progress;
  final Color ink;
  final Color gold;

  @override
  void paint(Canvas canvas, Size size) {
    final path = _starPath(size);

    // Every star is visible, selected or not — only the gold fill draws in.
    canvas.save();
    canvas.translate(1, 1);
    canvas.drawPath(
      path,
      Paint()
        ..color = ink.withValues(alpha: 0.3)
        ..style = PaintingStyle.stroke
        ..strokeWidth = 1.4
        ..strokeJoin = StrokeJoin.round,
    );
    canvas.restore();

    if (progress > 0) {
      canvas.drawPath(
        path,
        Paint()
          ..color = gold.withValues(alpha: 0.55 * progress)
          ..style = PaintingStyle.fill,
      );

      // A fat pencil stroke walks around the star, filling it in.
      final fill = Paint()
        ..color = gold
        ..style = PaintingStyle.stroke
        ..strokeWidth = size.shortestSide * 0.16
        ..strokeCap = StrokeCap.round
        ..strokeJoin = StrokeJoin.round;
      for (final metric in path.computeMetrics()) {
        canvas.drawPath(
          metric.extractPath(0, metric.length * progress),
          fill,
        );
      }
    }

    canvas.drawPath(
      path,
      Paint()
        ..color = ink
        ..style = PaintingStyle.stroke
        ..strokeWidth = 2.4
        ..strokeJoin = StrokeJoin.round,
    );
  }

  Path _starPath(Size size) {
    final center = Offset(size.width / 2, size.height / 2);
    final outer = size.shortestSide * 0.44;
    final inner = outer * 0.45;
    final path = Path();

    for (var i = 0; i < 10; i++) {
      final angle = -math.pi / 2 + i * math.pi / 5;
      final radius = i.isEven ? outer : inner;
      final wobble = 1 + 0.04 * math.sin(i * 2.7);
      final point = center +
          Offset(
            math.cos(angle) * radius * wobble,
            math.sin(angle) * radius * wobble,
          );
      if (i == 0) {
        path.moveTo(point.dx, point.dy);
      } else {
        path.lineTo(point.dx, point.dy);
      }
    }

    return path..close();
  }

  @override
  bool shouldRepaint(covariant _StarPainter oldDelegate) =>
      oldDelegate.progress != progress ||
      oldDelegate.ink != ink ||
      oldDelegate.gold != gold;
}

import 'package:flutter/material.dart';

/// An injectable animated-visual backend (I30, §5.1).
///
/// Every backend must be able to render the supplied [fallback] painter —
/// a kiosk must never show a broken or empty animation asset.
abstract interface class SketchAnimationBackend {
  Widget build(
    BuildContext context, {
    required Widget fallback,
    required bool active,
  });
}

/// Pure no-op: renders the painter unchanged. Use in tests and anywhere an
/// animation is not wanted (no loaders, no tickers, no timers).
final class NoOpSketchAnimationBackend implements SketchAnimationBackend {
  const NoOpSketchAnimationBackend();

  @override
  Widget build(
    BuildContext context, {
    required Widget fallback,
    required bool active,
  }) =>
      fallback;
}

/// The production default while real `.riv` art is unavailable: the
/// hand-drawn painter itself gets a small animated response to `active`.
final class FallbackSketchAnimationBackend implements SketchAnimationBackend {
  const FallbackSketchAnimationBackend();

  @override
  Widget build(
    BuildContext context, {
    required Widget fallback,
    required bool active,
  }) {
    return AnimatedScale(
      scale: active ? 1.06 : 1.0,
      duration: const Duration(milliseconds: 250),
      curve: Curves.easeOutBack,
      child: AnimatedOpacity(
        opacity: active ? 1 : 0.85,
        duration: const Duration(milliseconds: 250),
        child: fallback,
      ),
    );
  }
}

/// A sketch visual with two states and a guaranteed `CustomPainter` fallback.
///
/// The backend is injected so tests can be hermetic and a missing or broken
/// `.riv` asset can never leave a blank kiosk screen.
class SketchAnimation extends StatelessWidget {
  const SketchAnimation({
    super.key,
    required this.painter,
    this.backend = const FallbackSketchAnimationBackend(),
    this.active = false,
    this.fallbackKey,
    this.size,
  });

  /// The always-available hand-drawn visual.
  final CustomPainter painter;

  /// How the visual is animated; defaults to the painter fallback.
  final SketchAnimationBackend backend;

  /// Current state (e.g. a selected face or a waving character).
  final bool active;

  /// Key preserved on the fallback painter for existing component tests.
  final Key? fallbackKey;

  /// Optional square size constraint.
  final double? size;

  @override
  Widget build(BuildContext context) {
    Widget fallback = CustomPaint(key: fallbackKey, painter: painter);
    if (size != null) {
      fallback = SizedBox.square(dimension: size!, child: fallback);
    }

    try {
      return backend.build(context, fallback: fallback, active: active);
    } catch (_) {
      // A hostile/buggy backend cannot break the kiosk.
      return fallback;
    }
  }
}

import 'dart:math' as math;

import 'package:flutter/material.dart';

/// Draws the sketch focus frame as two jittered pencil passes that "write"
/// themselves in as [progress] runs from 0 to 1 (§5.1, §2.3.2).
///
/// Deliberately in-house: `rough_notation`'s autoplay timer calls
/// `AnimationController.forward()` after dispose when focus moves quickly,
/// which is a real race on a D-pad. This painter owns no timers — its owner
/// supplies the [Animation] (normally an `AnimationController`).
///
/// Shared by the emoji faces (I19) and the rating prompt (I23) so every
/// focusable kiosk control has the same unmistakable focus affordance.
class SketchFocusFramePainter extends CustomPainter {
  SketchFocusFramePainter({
    required this.progress,
    required this.color,
    required this.strokeWidth,
  }) : super(repaint: progress);

  final Animation<double> progress;
  final Color color;
  final double strokeWidth;

  @override
  void paint(Canvas canvas, Size size) {
    final t = progress.value;
    if (t <= 0) {
      return;
    }

    final paint = Paint()
      ..color = color
      ..style = PaintingStyle.stroke
      ..strokeWidth = strokeWidth
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;

    // Two passes, different jitter — a box drawn twice by hand.
    for (final seed in const [7, 23]) {
      final path = _jitteredRect(size, seed);
      for (final metric in path.computeMetrics()) {
        canvas.drawPath(metric.extractPath(0, metric.length * t), paint);
      }
    }
  }

  Path _jitteredRect(Size size, int seed) {
    final random = math.Random(seed);
    Offset corner(double x, double y) => Offset(
          x + random.nextDouble() * 5 - 2.5,
          y + random.nextDouble() * 5 - 2.5,
        );

    final topLeft = corner(0, 0);
    final topRight = corner(size.width, 0);
    final bottomRight = corner(size.width, size.height);
    final bottomLeft = corner(0, size.height);

    return Path()
      ..moveTo(topLeft.dx, topLeft.dy)
      ..lineTo(topRight.dx, topRight.dy)
      ..lineTo(bottomRight.dx, bottomRight.dy)
      ..lineTo(bottomLeft.dx, bottomLeft.dy)
      ..close();
  }

  @override
  bool shouldRepaint(covariant SketchFocusFramePainter oldDelegate) =>
      oldDelegate.color != color || oldDelegate.strokeWidth != strokeWidth;
}

/// Wobbly speech bubble with a tail pointing down at the character.
class _SpeechBubblePainter extends CustomPainter {
  const _SpeechBubblePainter({required this.ink, required this.paper});

  final Color ink;
  final Color paper;

  @override
  void paint(Canvas canvas, Size size) {
    final bubble = _bubblePath(size);

    canvas.drawPath(
      bubble,
      Paint()..color = paper..style = PaintingStyle.fill,
    );

    // A faint offset second pass keeps the outline hand-drawn.
    canvas.save();
    canvas.translate(1.5, 1);
    canvas.drawPath(
      bubble,
      Paint()
        ..color = ink.withValues(alpha: 0.28)
        ..style = PaintingStyle.stroke
        ..strokeWidth = 1.6
        ..strokeCap = StrokeCap.round
        ..strokeJoin = StrokeJoin.round,
    );
    canvas.restore();

    canvas.drawPath(
      bubble,
      Paint()
        ..color = ink
        ..style = PaintingStyle.stroke
        ..strokeWidth = 2.4
        ..strokeCap = StrokeCap.round
        ..strokeJoin = StrokeJoin.round,
    );
  }

  Path _bubblePath(Size size) {
    final w = size.width;
    final bottom = size.height - 12; // leave room for the tail
    return Path()
      ..moveTo(20, 5)
      ..lineTo(w - 22, 7)
      ..quadraticBezierTo(w - 3, 8, w - 5, 24)
      ..lineTo(w - 7, bottom - 20)
      ..quadraticBezierTo(w - 9, bottom - 2, w - 26, bottom - 4)
      ..lineTo(w * 0.58, bottom - 4)
      ..lineTo(w * 0.52, bottom + 10)
      ..lineTo(w * 0.46, bottom - 4)
      ..lineTo(24, bottom - 4)
      ..quadraticBezierTo(4, bottom - 6, 5, bottom - 24)
      ..lineTo(3, 24)
      ..quadraticBezierTo(4, 6, 20, 5)
      ..close();
  }

  @override
  bool shouldRepaint(covariant _SpeechBubblePainter oldDelegate) =>
      oldDelegate.ink != ink || oldDelegate.paper != paper;
}

/// Simple waving stick character — ink strokes, paper fill, warm cheeks.
class _CharacterPainter extends CustomPainter {
  const _CharacterPainter({
    required this.ink,
    required this.paper,
    required this.blush,
    required this.wave,
  });

  final Color ink;
  final Color paper;
  final Color blush;
  final Color wave;

  @override
  void paint(Canvas canvas, Size size) {
    final centerX = size.width / 2;

    final stroke = Paint()
      ..color = ink
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2.6
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;

    // Head.
    final head = Offset(centerX, size.height * 0.27);
    final radius = size.width * 0.18;
    canvas.drawCircle(
      head,
      radius,
      Paint()..color = paper..style = PaintingStyle.fill,
    );
    canvas.drawCircle(head, radius, stroke);

    // Eyes + smile.
    final eyeY = head.dy - radius * 0.1;
    canvas.drawCircle(
      head.translate(-radius * 0.36, 0),
      2.2,
      Paint()..color = ink..style = PaintingStyle.fill,
    );
    canvas.drawCircle(
      head.translate(radius * 0.36, 0),
      2.2,
      Paint()..color = ink..style = PaintingStyle.fill,
    );
    canvas.drawPath(
      Path()
        ..moveTo(head.dx - radius * 0.4, eyeY + radius * 0.35)
        ..quadraticBezierTo(
          head.dx,
          eyeY + radius * 0.85,
          head.dx + radius * 0.4,
          eyeY + radius * 0.35,
        ),
      stroke,
    );

    // Cheeks.
    final cheek = Paint()
      ..color = blush.withValues(alpha: 0.5)
      ..style = PaintingStyle.fill;
    canvas.drawCircle(
      head.translate(-radius * 0.72, radius * 0.35),
      3.4,
      cheek,
    );
    canvas.drawCircle(
      head.translate(radius * 0.72, radius * 0.35),
      3.4,
      cheek,
    );

    // Body, legs and the raised waving arm.
    final shoulder = Offset(centerX, size.height * 0.52);
    canvas.drawPath(
      Path()
        ..moveTo(shoulder.dx, shoulder.dy)
        ..quadraticBezierTo(
          shoulder.dx + 3,
          size.height * 0.66,
          shoulder.dx,
          size.height * 0.76,
        ),
      stroke,
    );
    canvas.drawLine(
      shoulder,
      Offset(centerX - size.width * 0.22, size.height * 0.66),
      stroke,
    );
    canvas.drawLine(
      shoulder,
      Offset(centerX + size.width * 0.26, size.height * 0.38),
      stroke,
    );
    canvas.drawLine(
      Offset(centerX, size.height * 0.76),
      Offset(centerX - size.width * 0.14, size.height * 0.94),
      stroke,
    );
    canvas.drawLine(
      Offset(centerX, size.height * 0.76),
      Offset(centerX + size.width * 0.14, size.height * 0.94),
      stroke,
    );

    // Little motion strokes around the waving hand.
    final motion = Paint()
      ..color = wave
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2
      ..strokeCap = StrokeCap.round;
    final handX = centerX + size.width * 0.26;
    final handY = size.height * 0.38;
    for (final offset in const [-7.0, 0.0, 7.0]) {
      canvas.drawLine(
        Offset(handX + offset, handY - 6),
        Offset(handX + offset + 2, handY - 12),
        motion,
      );
    }
  }

  @override
  bool shouldRepaint(covariant _CharacterPainter oldDelegate) =>
      oldDelegate.ink != ink ||
      oldDelegate.paper != paper ||
      oldDelegate.blush != blush ||
      oldDelegate.wave != wave;
}
