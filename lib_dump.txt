/// Pure state machine for one visitor's feedback action (§2.1.7).
///
/// No `BuildContext`, no widgets, no repository: just a submitter seam and
/// the §6.6 result vocabulary. All UI copy lives here as state messages so
/// the I24 screen never has to interpret a status code.
class FeedbackController extends Notifier<FeedbackState> {
  @override
  FeedbackState build() => const FeedbackIdle();

  /// Only idle (first choice) and error (retry) accept a selection.
  /// Success, cooldown, selecting and submitting are terminal until
  /// [reset] is called by the I24 prompt flow.
  bool get _canSelect => state is FeedbackIdle || state is FeedbackError;

  /// Walks `idle → selecting → submitting → success | cooldown | error`.
  ///
  /// The two first states are emitted synchronously before the first
  /// `await`, so the screen can show a pressed face while the durable write
  /// runs. Re-entrant calls are dropped before touching the submitter.
  Future<void> select(EmojiChoice choice) async {
    if (!_canSelect) {
      return;
    }

    state = FeedbackSelecting(choice);
    state = FeedbackSubmitting(choice);

    final SubmissionResult result;
    try {
      result = await ref.read(feedbackSubmitterProvider).submit(choice);
    } on FeedbackValidationException catch (error) {
      state = FeedbackError(error.message);
      return;
    } catch (_) {
      state = const FeedbackError('Something went wrong. Please try again.');
      return;
    }

    state = switch (result) {
      SubmissionAccepted() => FeedbackSuccess(choice),
      SubmissionRefused(status: SyncStatus.rejected) => const FeedbackCooldown(
          'Thanks for your feedback! Please wait a moment before trying again.',
        ),
      SubmissionRefused() => const FeedbackError(
          "We couldn't save your feedback. Please tell a staff member.",
        ),
    };
  }

  /// Returns the machine to `idle` for the next visitor / next question.
  void reset() => state = const FeedbackIdle();
}

        ),

          "We couldn't save your rating. Please tell a staff member.",

/// Starts the post-submit cooldown (§7 layer 1), implemented in the wiring
/// layer by an adapter over `CooldownService`.
abstract interface class FeedbackCooldownStarter {
  Future<void> start();
}

/// Invalid generated data (not a network problem): surfaced as user-facing
/// copy by the controller, distinct from the generic caught-exception error.
final class FeedbackValidationException implements Exception {
  const FeedbackValidationException(this.message);

  /// Safe to show to a visitor.
  final String message;

  @override
  String toString() => 'FeedbackValidationException($message)';
}

/// The controller's only view of submission — declared here in `domain/` so
/// the wiring layer can implement it without importing `presentation/`.
abstract interface class FeedbackSubmitter {
  Future<SubmissionResult> submit(EmojiChoice choice);
}

/// One visitor action: identity → entity → local-first submit → cooldown.
///
/// Pure Dart (§4.2): every side effect is an injected port. The repository
/// keeps the local-first promise — this use case never touches the network
/// directly and never blocks the visitor's thank-you on an upload.
class SubmitFeedback implements FeedbackSubmitter {
  SubmitFeedback({
    required this.repository,
    required this.deviceIds,
    required this.cooldown,
    required this.newClientId,
    DateTime Function()? now,
  }) : now = now ?? DateTime.now;

  final FeedbackRepository repository;
  final DeviceIdReader deviceIds;
  final FeedbackCooldownStarter cooldown;
  final ClientIdGenerator newClientId;

  /// Injectable clock; the entity requires UTC.
  final DateTime Function() now;

  @override
  Future<SubmissionResult> submit(EmojiChoice choice) async {
    final clientId = newClientId().trim();
    final deviceId = (await deviceIds.read()).trim();
    if (clientId.isEmpty || deviceId.isEmpty) {
      throw const FeedbackValidationException(
        "We couldn't prepare your feedback. Please try again.",
      );
    }

    final feedback = Feedback(
      clientId: clientId,
      deviceId: deviceId,
      emojiChoice: choice,
      createdAt: now().toUtc(),
    );

    final result = await repository.submit(feedback);

    // §7: lock the action after an accepted submission, and after a 429 so
    // a rejected visitor cannot hammer the server again. A 400 is a bug,
    // not a rate limit — leave the cooldown alone.
    if (result is SubmissionAccepted ||
        (result is SubmissionRefused &&
            result.status == SyncStatus.rejected)) {
      await cooldown.start();
    }

    return result;
  }
}

/// The rating controller's only view of submission (declared in `domain/`

abstract interface class RatingSubmitter {

/// One rating action: validate → identity → durable submit → cooldown.
///
/// Pure Dart; side effects are injected ports. The repository keeps the
/// local-first promise, so an offline rating is still a success.
class SubmitRating implements RatingSubmitter {
  SubmitRating({
    required this.repository,
    required this.deviceIds,
    required this.cooldown,
    required this.newClientId,
    DateTime Function()? now,
  }) : now = now ?? DateTime.now;

  final RatingRepository repository;
  final DeviceIdReader deviceIds;
  final RatingCooldownStarter cooldown;
  final ClientIdGenerator newClientId;
  final DateTime Function() now;

  @override
  Future<SubmissionResult> submit({
    required String projectId,
    required int score,
  }) async {
    if (projectId.trim().isEmpty || score < 1 || score > 5) {
      throw const RatingValidationException(
        "We couldn't save your rating. Please try again.",
      );
    }

    final clientId = newClientId().trim();
    final deviceId = (await deviceIds.read()).trim();
    if (clientId.isEmpty || deviceId.isEmpty) {
      throw const RatingValidationException(
        "We couldn't save your rating. Please try again.",
      );
    }

    final rating = Rating(
      clientId: clientId,
      deviceId: deviceId,
      projectId: projectId,
      score: score,
      createdAt: now().toUtc(),
    );

    final result = await repository.submit(rating);

    // Same cooldown policy as feedback (§7): accepted and 429, not 400.
    if (result is SubmissionAccepted ||
        (result is SubmissionRefused &&
            result.status == SyncStatus.rejected)) {
      await cooldown.start();
    }

    return result;
  }
}

/// Offline-first local queue — PROJECT_PROCESS.md §6.4, §6.6.
///
/// Both submission types share one database so the sync engine drains
/// them through a single connection and one schema version (§4.1 places
/// the queue under `features/feedback/data/local/`).
@DriftDatabase(tables: [PendingFeedback, PendingRatings])
class AppDatabase extends _$AppDatabase {
  /// Production database stored in the app documents directory.
  AppDatabase() : super(driftDatabase(name: 'ucstgo_kiosk'));

  /// Test/DI constructor — pass `NativeDatabase.memory()` for isolation,
  /// or a fake executor when unit-testing the data sources.
  AppDatabase.withExecutor(super.e);

  @override
  int get schemaVersion => 1;

  /// Store timestamps as ISO-8601 text instead of unix seconds.
  ///
  /// The domain requires UTC (§6.5) and the queue orders by `createdAt`;
  /// text storage round-trips the UTC flag instead of silently reading
  /// back local time. All writes are UTC, so lexicographic order = time
  /// order for the queue.
  @override
  DriftDatabaseOptions get options =>
      const DriftDatabaseOptions(storeDateTimeAsText: true);
}

/// Transport outcome of one queue-upload attempt — §6.6.
///
/// Shared by feedback and ratings so the sync engine needs one vocabulary.
/// Mapping HTTP codes happens once, in [mapHttpStatus], so Phase 6's real
/// datasources and the offline build behave identically.
enum RemoteOutcome {
  /// `200`/`2xx` or replay `409` — the row is safely on the server.
  success,

  /// `429` — server rate limit; terminal, never retried (§6.8).
  rejected,

  /// `400` (and other terminal 4xx) — a client/config bug, terminal.
  invalid,

  /// `5xx` — server-side failure; retry with backoff.
  serverError,

  /// Transport exception, timeout, redirect, or unexpected status; retry
  /// with backoff.
  networkFailure,
}

/// Pure HTTP-status → queue-outcome table (§6.6).
RemoteOutcome mapHttpStatus(int statusCode) {
  if ((statusCode >= 200 && statusCode < 300) || statusCode == 409) {
    return RemoteOutcome.success;
  }
  if (statusCode == 429) {
    return RemoteOutcome.rejected;
  }
  if (statusCode == 400) {
    return RemoteOutcome.invalid;
  }
  if (statusCode >= 500 && statusCode < 600) {
    return RemoteOutcome.serverError;
  }
  if (statusCode >= 400 && statusCode < 500) {
    // 401 / 403 / 404 … are configuration mistakes, not transient faults.
    return RemoteOutcome.invalid;
  }
  return RemoteOutcome.networkFailure;
}

/// Outcome of a local-first submit — PROJECT_PROCESS.md §4.2, §6.4, §6.6.
///
/// The UI must react identically whether the upload happened now
/// ([SyncStatus.synced]) or will happen later ([SyncStatus.pending]):

  /// Local status of the row after the submit attempt.
  final SyncStatus status;

/// The submission is durably stored and safe.
///
/// [status] is either [SyncStatus.pending] (offline / remote failed —

  final String reason;

/// Queue lifecycle for a submission — PROJECT_PROCESS.md §6.6.
///
/// Every local queue row carries one of these values. [pending] is the
/// only non-terminal state: the SyncService retries it with exponential
/// backoff until it becomes [synced] (200/409), [rejected] (429 — rate
/// limited, do not retry) or [failedPermanent] (400 — a client bug).
enum SyncStatus {
  /// Durable locally; upload pending or retry scheduled via nextRetryAt.
  pending('pending'),

  /// Server accepted (200) or replay detected (409) — terminal, row
  /// leaves the queue (§6.6).
  synced('synced'),

  /// Rate limited (429) — terminal, never retried (§6.8).
  rejected('rejected'),

  /// Malformed payload (400) — terminal; signals a client bug.
  failedPermanent('failed_permanent');

  const SyncStatus(this.wireValue);

  /// Exact string stored in Drift and referenced by the schema (§6.6).
  final String wireValue;

  /// True once the sync engine must stop retrying this row.
  bool get isTerminal => this != pending;

  /// Serializes for Drift/Postgres storage.
  String toJson() => wireValue;

  /// Parses a stored/transmitted value; unknown values are data bugs.
  static SyncStatus fromJson(String value) {
    for (final status in values) {
      if (status.wireValue == value) {
        return status;
      }
    }
    throw FormatException('Unknown SyncStatus wire value: "$value"');
  }
}

/// Typed CRUD over the durable feedback queue — PROJECT_PROCESS.md §6.4, §6.6.
///
/// This is the only place that maps [Feedback] entities to Drift rows. It
/// deliberately knows nothing about HTTP, Supabase or connectivity: the
/// repository (Iteration 11) orchestrates, the sync engine (Iteration 16)
/// walks the queue through [pending] + [applySyncStatus].
class FeedbackLocalDataSource {
  FeedbackLocalDataSource(this._db, {DateTime Function()? now})
      : _now = now ?? DateTime.now;

  final AppDatabase _db;

  /// Injectable clock so retry-deadline behavior is tested without waiting.
  final DateTime Function() _now;

  /// Persists a submission as `pending`, durably, before any network call.
  ///
  /// A duplicate [Feedback.clientId] is a bug (the idempotency key must be
  /// fresh per visitor action) and is rejected by the primary key.
  Future<void> insert(Feedback feedback) async {
    await _db.into(_db.pendingFeedback).insert(
          PendingFeedbackCompanion.insert(
            clientId: feedback.clientId,
            deviceId: feedback.deviceId,
            emojiChoice: feedback.emojiChoice.wireValue,
            createdAt: feedback.createdAt.toUtc(),
          ),
        );
  }

  /// Rows eligible for upload, FIFO by `createdAt`.
  ///
  /// "Eligible" means `pending` **and** past the backoff deadline: a row
  /// waiting for its next retry is not offered to the sync engine early.
  /// Retry-deadline filtering happens in Dart on purpose — comparing
  /// ISO-8601 text in SQLite is not reliably chronological for
  /// sub-millisecond timestamps, and DateTime comparison here is exact.
  Future<List<QueuedFeedback>> pending({int limit = 50}) async {
    assert(limit > 0, 'limit must be positive');

    final rows = await (_db.select(_db.pendingFeedback)
          ..where(
            (table) =>
                table.syncStatus.equals(SyncStatus.pending.wireValue),
          ))
        .get();

    final now = _now().toUtc();
    final due = rows.where((row) {
      final retryAt = row.nextRetryAt?.toUtc();
      return retryAt == null || !retryAt.isAfter(now);
    }).toList()
      ..sort((a, b) {
        final byTime = a.createdAt.compareTo(b.createdAt);
        return byTime != 0 ? byTime : a.clientId.compareTo(b.clientId);
      });

    return due.take(limit).map(_toQueued).toList();
  }

  /// Live count of rows still awaiting upload — includes rows waiting for
  /// their backoff deadline, unlike [pending].
  Stream<int> watchPendingCount() {
    final count = _db.pendingFeedback.clientId.count();
    final query = _db.selectOnly(_db.pendingFeedback)
      ..addColumns([count])
      ..where(
        _db.pendingFeedback.syncStatus.equals(SyncStatus.pending.wireValue),
      );

    return query.map((row) => row.read(count) ?? 0).watchSingle();
  }

  /// Terminal success (200 or replay 409) — §6.6.
  Future<void> markSynced(String clientId) =>
      _setStatus(clientId, SyncStatus.synced);

  /// Rate limited by the server (429) — terminal, never retried.
  Future<void> markRejected(String clientId) =>
      _setStatus(clientId, SyncStatus.rejected);

  /// Malformed payload (400) — terminal, signals a client bug.
  Future<void> markFailedPermanent(String clientId) =>
      _setStatus(clientId, SyncStatus.failedPermanent);

  /// Records a failed attempt and blocks retries until [nextRetryAt].
  ///
  /// Increments [PendingFeedback.attempts] so the backoff policy in
  /// Iteration 16 can cap at six retries.
  Future<void> scheduleRetry(String clientId, DateTime nextRetryAt) async {
    final row = await (_db.select(_db.pendingFeedback)
          ..where((table) => table.clientId.equals(clientId)))
        .getSingle();

    await (_db.update(_db.pendingFeedback)
          ..where((table) => table.clientId.equals(clientId)))
        .write(
      PendingFeedbackCompanion(
        attempts: Value(row.attempts + 1),
        lastAttemptAt: Value(_now().toUtc()),
        nextRetryAt: Value(nextRetryAt.toUtc()),
      ),
    );
  }

  /// Removes a row (after a successful sync, if the queue is trimmed).
  Future<void> delete(String clientId) async {
    await (_db.delete(_db.pendingFeedback)
          ..where((table) => table.clientId.equals(clientId)))
        .go();
  }

  Future<void> _setStatus(String clientId, SyncStatus status) async {
    await (_db.update(_db.pendingFeedback)
          ..where((table) => table.clientId.equals(clientId)))
        .write(
      PendingFeedbackCompanion(
        syncStatus: Value(status.wireValue),
        nextRetryAt: const Value(null),
      ),
    );
  }

  QueuedFeedback _toQueued(PendingFeedbackData row) => QueuedFeedback(
        feedback: _toEntity(row),
        attempts: row.attempts,
      );

  Feedback _toEntity(PendingFeedbackData row) {
    return Feedback(
      clientId: row.clientId,
      deviceId: row.deviceId,
      emojiChoice: EmojiChoice.fromJson(row.emojiChoice),
      createdAt: row.createdAt.toUtc(),
    );
  }

/// Local-first orchestration — PROJECT_PROCESS.md §6.4, §6.6.
///
/// The contract in one sentence: **the submission is durable before any
/// network work starts, and a network problem never fails the visitor's
/// action.** Only a definitive server refusal (429 / 400) is surfaced,
/// and even those rows stay in the queue with a terminal status.
class FeedbackRepositoryImpl implements FeedbackRepository {
  FeedbackRepositoryImpl({
    required this.local,
    required this.remote,
    required this.isOnline,
  });

  final FeedbackLocalDataSource local;
  final FeedbackRemoteDataSource remote;

  /// Injected connectivity check; Phase 2 wires `ConnectivityService`
  /// here, tests inject a fake.
  final Future<bool> Function() isOnline;

  @override
  Future<SubmissionResult> submit(Feedback feedback) async {
    // 1. Durable write first — the queue is the source of truth (§6.4).
    await local.insert(feedback);

    // 2. Best-effort upload, only when connectivity is believed usable.
    if (!await isOnline()) {
      return const SubmissionAccepted(SyncStatus.pending);
    }

    final RemoteOutcome outcome;
    try {
      // The seam takes the exact §6.2 wire DTO, so Phase 6 cannot bypass
      // the payload contract.
      outcome = await remote.submit(FeedbackModel.fromFeedback(feedback));
    } catch (_) {
      // The seam promises not to throw; stay defensive so a buggy
      // datasource can never break the visitor's thank-you.
      return const SubmissionAccepted(SyncStatus.pending);
    }

    switch (outcome) {
      case RemoteOutcome.success:
        await local.markSynced(feedback.clientId);
        return const SubmissionAccepted(SyncStatus.synced);

      case RemoteOutcome.rejected:
        await local.markRejected(feedback.clientId);
        return const SubmissionRefused(
          SyncStatus.rejected,
          'rate limited by the server',
        );

      case RemoteOutcome.invalid:
        await local.markFailedPermanent(feedback.clientId);
        return const SubmissionRefused(
          SyncStatus.failedPermanent,
          'payload rejected by the server',
        );

      case RemoteOutcome.serverError:
      case RemoteOutcome.networkFailure:
        // Row stays pending; SyncService retries with backoff (§6.6).
        return const SubmissionAccepted(SyncStatus.pending);
    }
  }

  @override
  Future<List<QueuedFeedback>> pending({int limit = 50}) =>
      local.pending(limit: limit);

  @override
  Stream<int> watchPendingCount() => local.watchPendingCount();

  @override
  Future<void> applySyncStatus(
    String clientId,
    SyncStatus status, {
    DateTime? nextRetryAt,
  }) async {
    switch (status) {
      case SyncStatus.synced:
        await local.markSynced(clientId);
      case SyncStatus.rejected:
        await local.markRejected(clientId);
      case SyncStatus.failedPermanent:
        await local.markFailedPermanent(clientId);
      case SyncStatus.pending:
        if (nextRetryAt != null) {
          await local.scheduleRetry(clientId, nextRetryAt);
        }
    }
  }
}

import 'dart:async';
import 'dart:io';

import 'package:connectivity_plus/connectivity_plus.dart';

/// Physical-link seam — "is a network interface attached?" (§6.7).
///
/// This deliberately says **nothing** about internet access: a Wi-Fi icon
/// can be lit while the kiosk sits behind a captive portal. Link state is
/// only the cheap first question; [ReachabilityProbe] asks the second.
abstract interface class LinkStatusSource {
  /// One-shot link check. Implementations never throw.
  Future<bool> hasLink();

  /// Emits `true`/`false` whenever the physical link changes.
  Stream<bool> get onLinkChange;
}

/// Production link source backed by `connectivity_plus`.
class ConnectivityPlusLinkSource implements LinkStatusSource {
  ConnectivityPlusLinkSource({Connectivity? connectivity})
      : _connectivity = connectivity ?? Connectivity();

  final Connectivity _connectivity;

  /// Only `none` means the link is down; any transport counts as up.
  static bool hasUsableLink(List<ConnectivityResult> results) =>
      results.any((result) => result != ConnectivityResult.none);

  @override
  Future<bool> hasLink() async {
    try {
      return hasUsableLink(await _connectivity.checkConnectivity());
    } catch (_) {
      // A plugin failure is reported as "no link" — sync stays paused.
      return false;
    }
  }

  @override
  Stream<bool> get onLinkChange =>
      _connectivity.onConnectivityChanged.map(hasUsableLink);
}

/// Reachability seam — can we actually reach the backend? (§6.7).
abstract interface class ReachabilityProbe {
  /// `true` when the backend answered. Implementations never throw.
  Future<bool> probe();

  /// Releases any underlying resources (e.g. the HTTP client).
  void dispose();
}

/// Production probe: `HEAD` to the configured backend base URL.
///
/// **Captive-portal note:** a completed TLS response from the real host is
/// the proof we need. A portal that hijacks the connection either fails TLS
/// or returns a non-2xx/3xx page; either way the kiosk stays offline and
/// submissions keep queuing locally instead of vanishing.
///
/// With no `SUPABASE_URL` configured (Phases 1–5) the probe is link-only:
/// there is nothing to upload to, so a live link is the best answer.
class HttpReachabilityProbe implements ReachabilityProbe {
  HttpReachabilityProbe({
    required String baseUrl,
    this.timeout = const Duration(seconds: 5),
    HttpClient? client,
  })  : baseUrl = baseUrl.trim(),
        _client = client ?? HttpClient();

  final String baseUrl;
  final Duration timeout;
  final HttpClient _client;

  @override
  Future<bool> probe() async {
    if (baseUrl.isEmpty) {
      return true;
    }

    try {
      final request = await _client.headUrl(Uri.parse(baseUrl)).timeout(timeout);
      // A portal redirect is not the backend talking — treat it as failure.
      request.followRedirects = false;
      final response = await request.close().timeout(timeout);
      await response.drain<void>().timeout(timeout);
      // Any non-5xx answer — including 401 from an unauthenticated HEAD —
      // proves the host is alive and reachable.
      return response.statusCode < 500;
    } catch (_) {
      return false;
    }
  }

  @override
  void dispose() => _client.close(force: true);
}

/// The slice of connectivity the sync scheduler depends on.
///
/// Kept narrow so trigger tests can fake it without a link source or probe.
abstract interface class ConnectivityStatus {
  /// Last known answer; `false` until [start] runs.
  bool get isOnline;

  /// Change-only stream of online/offline edges.
  Stream<bool> get online;

  /// Subscribes to link changes, probes once, and returns the result.
  Future<bool> start();
}

/// Combines link state and reachability into one trusted `online` signal.
///
/// Rules (§6.7):
/// - Link down → **offline immediately**, no probe (saves the 5s timeout).
/// - Link up → probe. Probe fails → offline, even though Wi-Fi says yes.
/// - Link change → probe again; a rising edge is what unblocks the queue.
/// - Concurrent [checkNow] calls share one in-flight check, so a burst of
///   triggers cannot stampede the network.
class ConnectivityService implements ConnectivityStatus {
  ConnectivityService({
    required this.link,
    required this.probe,
  });

  final LinkStatusSource link;
  final ReachabilityProbe probe;

  final _onlineController = StreamController<bool>.broadcast();

  StreamSubscription<bool>? _linkSubscription;
  Future<bool>? _inFlight;
  bool _online = false;
  bool _linkUp = false;
  bool _disposed = false;

  /// Last known answer. Starts `false` until [start] or [checkNow] runs.
  @override
  bool get isOnline => _online;

  /// Emits only on **changes** (rising and falling edges), never replays.
  /// Pair it with [isOnline] for the current value.
  @override
  Stream<bool> get online => _onlineController.stream;

  /// Begins observing link changes and performs the first check.
  ///
  /// Idempotent: calling it twice keeps one subscription and coalesces the
  /// checks.
  @override
  Future<bool> start() {
    if (_disposed) {
      return Future.value(false);
    }
    _linkSubscription ??= link.onLinkChange.listen(_onLinkChange);
    return checkNow();
  }

  /// Runs one link + reachability check now.
  ///
  /// Concurrent callers share the same future, so the scheduler can fire
  /// several triggers at once without producing several probes.
  Future<bool> checkNow() {
    if (_disposed) {
      return Future.value(false);
    }
    return _inFlight ??= _check().whenComplete(() => _inFlight = null);
  }

  Future<bool> _check() async {
    final hasLink = await _safeHasLink();
    _linkUp = hasLink;
    if (!hasLink) {
      return _setOnline(false);
    }

    final reachable = await _safeProbe();
    // The link may have dropped while the probe was in flight; never
    // publish "online" from a stale probe.
    if (!_linkUp) {
      return _setOnline(false);
    }
    return _setOnline(reachable);
  }

  Future<bool> _safeHasLink() async {
    try {
      return await link.hasLink();
    } catch (_) {
      return false;
    }
  }

  Future<bool> _safeProbe() async {
    try {
      return await probe.probe();
    } catch (_) {
      return false;
    }
  }

  void _onLinkChange(bool hasLink) {
    if (_disposed) {
      return;
    }
    _linkUp = hasLink;
    if (!hasLink) {
      _setOnline(false);
      return;
    }
    unawaited(checkNow());
  }

  bool _setOnline(bool value) {
    if (_online == value) {
      return value;
    }
    _online = value;
    if (!_disposed) {
      _onlineController.add(value);
    }
    return value;
  }

  /// Stops observing, closes the stream, and releases the probe.
  ///
  /// Ownership note: the service owns the [probe] but not the [link] source,
  /// so the link's stream is left for its creator.
  Future<void> dispose() async {
    if (_disposed) {
      return;
    }
    _disposed = true;
    await _linkSubscription?.cancel();
    _linkSubscription = null;
    await _onlineController.close();
    probe.dispose();
  }
}

import 'package:shared_preferences/shared_preferences.dart';

/// The two independently rate-limited visitor actions (§7).
enum CooldownAction {
  feedback('feedback'),
  rating('rating');

  const CooldownAction(this.storageKey);

  /// Suffix used in the persisted key; never rename (would orphan deadlines).
  final String storageKey;
}

/// Persistence seam for cooldown deadlines — injectable for tests.
abstract interface class CooldownStorage {
  Future<DateTime?> readDeadline(CooldownAction action);

  Future<void> writeDeadline(CooldownAction action, DateTime deadline);

  Future<void> clearDeadline(CooldownAction action);
}

/// Production storage: one ISO-8601 UTC deadline per action (§7).
class SharedPreferencesCooldownStorage implements CooldownStorage {
  const SharedPreferencesCooldownStorage();

  static String keyFor(CooldownAction action) =>
      'cooldown_deadline_${action.storageKey}';

  @override
  Future<DateTime?> readDeadline(CooldownAction action) async {
    final preferences = await SharedPreferences.getInstance();
    final raw = preferences.getString(keyFor(action));
    if (raw == null) {
      return null;
    }
    // Corrupt values are treated as "no cooldown" — never lock the kiosk.
    return DateTime.tryParse(raw)?.toUtc();
  }

  @override
  Future<void> writeDeadline(CooldownAction action, DateTime deadline) async {
    final preferences = await SharedPreferences.getInstance();
    await preferences.setString(
      keyFor(action),
      deadline.toUtc().toIso8601String(),
    );
  }

  @override
  Future<void> clearDeadline(CooldownAction action) async {
    final preferences = await SharedPreferences.getInstance();
    await preferences.remove(keyFor(action));
  }
}

/// Client-side UX cooldown, 30s per action by default — §7 layer 1.
///
/// This is **not** security: the server rate limit (§6.8) stays the law.
/// The cooldown only stops rapid repeat taps and survives restarts through
/// a persisted deadline, so killing the app does not reset it.
///
/// The clock and storage are injected so the 30s window is tested in
/// milliseconds instead of real time.
class CooldownService {
  CooldownService({
    required this.storage,
    this.duration = const Duration(seconds: 30),
    DateTime Function()? now,
  }) : now = now ?? DateTime.now;

  final CooldownStorage storage;

  /// Default 30s, tunable per §13.
  final Duration duration;

  final DateTime Function() now;

  /// True when the action has no active cooldown.
  Future<bool> canSubmit(CooldownAction action) async =>
      (await timeRemaining(action)) == Duration.zero;

  /// Time left before [action] becomes available again.
  ///
  /// Capped at [duration] so a backwards clock jump cannot lock the kiosk
  /// for longer than one cooldown window.
  Future<Duration> timeRemaining(CooldownAction action) async {
    final deadline = await storage.readDeadline(action);
    if (deadline == null) {
      return Duration.zero;
    }

    final remaining = deadline.toUtc().difference(now().toUtc());
    if (remaining.isNegative) {
      return Duration.zero;
    }
    return remaining > duration ? duration : remaining;
  }

  /// Starts the cooldown for [action] after a successful submission.
  Future<void> start(CooldownAction action) async {
    await storage.writeDeadline(
      action,
      now().toUtc().add(duration),
    );
  }

  /// Ends the cooldown immediately (operator/testing escape hatch).
  Future<void> clear(CooldownAction action) => storage.clearDeadline(action);

  /// Emits the countdown until [action] is available, then completes.
  ///
  /// The first value is emitted immediately so a `StreamBuilder` can show
  /// the current state without an empty frame.
  Stream<Duration> remaining(
    CooldownAction action, {
    Duration tick = const Duration(seconds: 1),
  }) async* {
    while (true) {
      final left = await timeRemaining(action);
      yield left;
      if (left == Duration.zero) {
        return;
      }
      await Future<void>.delayed(tick);
    }
  }
}

import 'package:shared_preferences/shared_preferences.dart';
import 'package:uuid/uuid.dart';

/// Persistence seam for the anonymous device id — injectable for tests.
abstract interface class DeviceIdStorage {
  Future<String?> read();

  Future<void> write(String value);
}

/// Production storage: `shared_preferences` (§6.3).
class SharedPreferencesDeviceIdStorage implements DeviceIdStorage {
  const SharedPreferencesDeviceIdStorage();

  /// Stable key; changing it would orphan every existing device identity.
  static const storageKey = 'device_id';

  @override
  Future<String?> read() async {
    final preferences = await SharedPreferences.getInstance();
    return preferences.getString(storageKey);
  }

  @override
  Future<void> write(String value) async {
    final preferences = await SharedPreferences.getInstance();
    await preferences.setString(storageKey, value);
  }
}

/// Anonymous, stable device identity — PROJECT_PROCESS.md §6.3, §2.4.
///
/// Generated **once** on first launch as a UUID v4 and reused for the
/// device's lifetime. There is no account, no name, no PII — just an
/// opaque id that lets the server apply per-device rate limits (§6.8).
///
/// The id is cached in memory after the first read, and concurrent first
/// calls coalesce into a single generation + write (two widgets requesting
/// identity at startup must not create two ids).
class DeviceIdService {
  DeviceIdService({
    required this.storage,
    Uuid? uuid,
  }) : uuid = uuid ?? const Uuid();

  final DeviceIdStorage storage;
  final Uuid uuid;

  String? _cached;
  Future<String>? _inFlight;

  /// Returns the persisted device id, creating it on first use.
  Future<String> getOrCreate() {
    final cached = _cached;
    if (cached != null) {
      return Future.value(cached);
    }
    return _inFlight ??= _loadOrCreate().whenComplete(() => _inFlight = null);
  }

  Future<String> _loadOrCreate() async {
    final stored = await storage.read();
    if (stored != null && stored.trim().isNotEmpty) {
      _cached = stored;
      return stored;
    }

    final created = uuid.v4();
    await storage.write(created);
    _cached = created;
    return created;
  }
}

import 'dart:async';

import './connectivity_service.dart';
import './sync_service.dart';

/// Wires the four §6.4 sync triggers into one coalesced background sweep.
///
/// Triggers:
/// 1. [start] — app launch, after connectivity's first probe.
/// 2. [ConnectivityStatus.online] rising edge — the network came back.
/// 3. A periodic timer (5 minutes by default).
/// 4. [notifyLocalWrite] — a debounced nudge after each durable save.
///
/// Coalescing has two layers: repeated nudges share one timer here, and
/// overlapping sweeps collapse inside [SyncRunner.syncAll]. No trigger
/// starts a sweep while connectivity reports offline; the rising edge is
/// the recovery path.
class SyncScheduler {
  SyncScheduler({
    required this.sync,
    required this.connectivity,
    this.interval = const Duration(minutes: 5),
    this.nudgeDelay = const Duration(seconds: 1),
  })  : assert(interval > Duration.zero, 'interval must be positive'),
        assert(nudgeDelay >= Duration.zero, 'nudgeDelay must not be negative');

  final SyncRunner sync;
  final ConnectivityStatus connectivity;

  /// Periodic sweep cadence (§13: 5 minutes).
  final Duration interval;

  /// Post-write debounce window (§6.4: ~1 second).
  final Duration nudgeDelay;

  StreamSubscription<bool>? _connectivitySubscription;
  Timer? _periodicTimer;
  Timer? _nudgeTimer;
  bool _started = false;
  bool _disposed = false;

  /// Starts observing triggers and performs the app-start sweep.
  ///
  /// Bootstrap may call this unawaited: the returned future completes after
  /// the initial sweep, never at the end of the scheduler's life.
  Future<void> start() async {
    if (_started || _disposed) {
      return;
    }
    _started = true;

    _connectivitySubscription = connectivity.online.listen(
      _onConnectivityChange,
    );
    _periodicTimer = Timer.periodic(
      interval,
      (_) => unawaited(_trigger()),
    );

    final online = await connectivity.start();
    if (online && !_disposed) {
      await sync.syncAll();
    }
  }

  /// Requests a sweep shortly after a durable local write.
  ///
  /// Repeated calls inside [nudgeDelay] collapse into one timer. Other
  /// triggers do **not** cancel a pending nudge: a sweep that started
  /// before the write may already have queried the queue, so the nudge
  /// remains the safety net that picks the new row up.
  void notifyLocalWrite() {
    if (_disposed) {
      return;
    }
    _nudgeTimer?.cancel();
    _nudgeTimer = Timer(nudgeDelay, () => unawaited(_trigger()));
  }

  void _onConnectivityChange(bool online) {
    if (online) {
      unawaited(_trigger());
    }
  }

  Future<void> _trigger() async {
    if (_disposed) {
      return;
    }
    if (!connectivity.isOnline) {
      return;
    }
    await sync.syncAll();
  }

  void _cancelNudge() {
    _nudgeTimer?.cancel();
    _nudgeTimer = null;
  }

  /// Stops all timers and the link subscription. Idempotent.
  Future<void> dispose() async {
    if (_disposed) {
      return;
    }
    _disposed = true;
    _cancelNudge();
    _periodicTimer?.cancel();
    _periodicTimer = null;

    final subscription = _connectivitySubscription;
    _connectivitySubscription = null;
    await subscription?.cancel();
  }
}

/// The scheduler's view of the sync engine.
abstract interface class SyncRunner {
  /// Drains every due row; concurrent calls coalesce in the implementation.
  Future<void> syncAll();
}

/// Drains the durable queues through the tested remote seams — §6.4/§6.6/§13.
///
/// Design rules:
/// - **Serial:** one upload at a time; a low-end kiosk should never have
///   several requests in flight.
/// - **Both queues:** feedback first, then ratings, in one coalesced sweep
///   (I28). The rating seams are optional so the I16 tests still compile.
/// - **Batched but complete:** `pending(limit:)` bounds each query; the loop
///   repeats until nothing is due.
/// - **Coalesced:** concurrent [syncAll] calls share one run.
/// - **Durable backoff:** attempt counts come from the rows, not memory, so
///   killing the app does not reset the retry ladder.
class SyncService implements SyncRunner {
  SyncService({
    required this.queue,
    required this.remote,
    this.ratingQueue,
    this.ratingRemote,
    DateTime Function()? now,
    this.batchSize = 50,
    this.retryDelays = defaultRetryDelays,
  })  : assert(batchSize > 0, 'batchSize must be positive'),
        now = now ?? DateTime.now;

  /// §13: 2 → 4 → 8 → 16 → 32 → 60s, then terminal.
  static const defaultRetryDelays = <Duration>[
    Duration(seconds: 2),
    Duration(seconds: 4),
    Duration(seconds: 8),
    Duration(seconds: 16),
    Duration(seconds: 32),
    Duration(seconds: 60),
  ];

  final FeedbackRepository queue;
  final FeedbackRemoteDataSource remote;

  /// Rating queue/remote — both must be supplied to drain ratings (I28).
  final RatingRepository? ratingQueue;
  final RatingRemoteDataSource? ratingRemote;

  final DateTime Function() now;
  final int batchSize;
  final List<Duration> retryDelays;

  Future<void>? _inFlight;

  /// Live feedback count, including backoff-waiting rows.
  Stream<int> get pendingCount => queue.watchPendingCount();

  @override
  Future<void> syncAll() {
    return _inFlight ??= _drain().whenComplete(() => _inFlight = null);
  }

  Future<void> _drain() async {
    await _drainQueue(
      load: () => queue.pending(limit: batchSize),
      handle: _syncFeedback,
    );

    final ratings = ratingQueue;
    final ratingUpload = ratingRemote;
    if (ratings != null && ratingUpload != null) {
      await _drainQueue(
        load: () => ratings.pending(limit: batchSize),
        handle: _syncRating,
      );
    }
  }

  Future<void> _drainQueue<T>({
    required Future<List<T>> Function() load,
    required Future<void> Function(T) handle,
  }) async {
    while (true) {
      final batch = await load();
      if (batch.isEmpty) {
        return;
      }
      for (final item in batch) {
        await handle(item);
      }
    }
  }

  Future<void> _syncFeedback(QueuedFeedback item) => _submitAndApply(
        clientId: item.feedback.clientId,
        attempts: item.attempts,
        submit: () => remote.submit(FeedbackModel.fromFeedback(item.feedback)),
        apply: (status, {nextRetryAt}) => queue.applySyncStatus(
          item.feedback.clientId,
          status,
          nextRetryAt: nextRetryAt,
        ),
      );

  Future<void> _syncRating(QueuedRating item) => _submitAndApply(
        clientId: item.rating.clientId,
        attempts: item.attempts,
        submit: () => ratingRemote!.submit(RatingModel.fromRating(item.rating)),
        apply: (status, {nextRetryAt}) => ratingQueue!.applySyncStatus(
          item.rating.clientId,
          status,
          nextRetryAt: nextRetryAt,
        ),
      );

  /// One item's §6.6 outcome matrix + §13 backoff, shared by both queues.
  Future<void> _submitAndApply({
    required String clientId,
    required int attempts,
    required Future<RemoteOutcome> Function() submit,
    required Future<void> Function(SyncStatus status, {DateTime? nextRetryAt})
        apply,
  }) async {
    final outcome = await _safeSubmit(submit);

    switch (outcome) {
      case RemoteOutcome.success:
        await apply(SyncStatus.synced);

      case RemoteOutcome.rejected:
        await apply(SyncStatus.rejected);

      case RemoteOutcome.invalid:
        await apply(SyncStatus.failedPermanent);

      case RemoteOutcome.serverError:
      case RemoteOutcome.networkFailure:
        if (attempts >= retryDelays.length) {
          // Six retries exhausted — stop burning battery on a hopeless row.
          await apply(SyncStatus.failedPermanent);
          return;
        }
        await apply(
          SyncStatus.pending,
          nextRetryAt: now().toUtc().add(retryDelays[attempts]),
        );
    }
  }

  Future<RemoteOutcome> _safeSubmit(
    Future<RemoteOutcome> Function() submit,
  ) async {
    try {
      return await submit();
    } catch (_) {
      // A buggy datasource cannot abort the whole sweep.
      return RemoteOutcome.networkFailure;
    }
  }
}

/// A wobbly line that draws itself from left to right, looping forever.
class _PencilUnderlinePainter extends CustomPainter {
  const _PencilUnderlinePainter({
    required this.progress,
    required this.color,
  });

  final double progress;
  final Color color;

  @override
  void paint(Canvas canvas, Size size) {
    if (progress <= 0) {
      return;
    }

    final end = size.width * progress;
    final path = Path();
    const steps = 24;
    for (var step = 0; step <= steps; step++) {
      final x = end * step / steps;
      final y = size.height / 2 + (step.isEven ? 1.5 : -1.5);
      if (step == 0) {
        path.moveTo(x, y);
      } else {
        path.lineTo(x, y);
      }
    }

    canvas.drawPath(
      path,
      Paint()
        ..color = color
        ..style = PaintingStyle.stroke
        ..strokeWidth = 2.5
        ..strokeCap = StrokeCap.round,
    );
  }

  @override
  bool shouldRepaint(covariant _PencilUnderlinePainter oldDelegate) =>
      oldDelegate.progress != progress || oldDelegate.color != color;
}

class _CooldownScreenState extends State<CooldownScreen> {
  StreamSubscription<Duration>? _subscription;
  Duration? _remaining;
  bool _expired = false;

  @override
  void initState() {
    super.initState();
    _listen(widget.remaining);
  }

  @override
  void didUpdateWidget(covariant CooldownScreen oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (oldWidget.remaining != widget.remaining) {
      _subscription?.cancel();
      _listen(widget.remaining);
    }

  @override
  void dispose() {
    _subscription?.cancel();
    super.dispose();

  void _listen(Stream<Duration> stream) {
    _remaining = null;
    _expired = false;
    _subscription = stream.listen(
      (left) {
        if (!mounted) {
          return;
        }
        setState(() => _remaining = left);
        if (left <= Duration.zero) {
          _fireExpiry();
        }
      },
      onDone: _fireExpiry,
      onError: (_) => _fireExpiry(),
    );
  }

  void _fireExpiry() {
    if (_expired) {
      return;
    }
    _expired = true;
    widget.onExpiry?.call();
  }

  Widget build(BuildContext context) {

    final seconds = left?.inSeconds;

/// Wobbly clock face with a hand that unwinds as the deadline approaches.
class _ClockPainter extends CustomPainter {
  const _ClockPainter({
    required this.ink,
    required this.glow,
    required this.fraction,
  });

  final Color ink;
  final Color glow;
  final double fraction;

  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height / 2);
    final radius = size.shortestSide * 0.42;

    canvas.drawCircle(
      center,
      radius,
      Paint()..color = glow.withValues(alpha: 0.25),
    );

    final stroke = Paint()
      ..color = ink
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2.6
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;

    canvas.drawCircle(center, radius, stroke);

    for (var tick = 0; tick < 12; tick++) {
      final angle = tick * 2 * math.pi / 12;
      final inner = radius * 0.82;
      final outer = radius * 0.92;
      canvas.drawLine(
        center + Offset(math.cos(angle) * inner, math.sin(angle) * inner),
        center + Offset(math.cos(angle) * outer, math.sin(angle) * outer),
        stroke..strokeWidth = 1.6,
      );
    }

    final handAngle = -math.pi / 2 + (1 - fraction) * 2 * math.pi;
    canvas.drawLine(
      center,
      center +
          Offset(math.cos(handAngle) * radius * 0.6,
              math.sin(handAngle) * radius * 0.6),
      stroke
        ..strokeWidth = 3
        ..color = ink,
    );
    canvas.drawCircle(center, 4, Paint()..color = ink);
  }

  @override
  bool shouldRepaint(covariant _ClockPainter oldDelegate) =>
      oldDelegate.fraction != fraction ||
      oldDelegate.ink != ink ||
      oldDelegate.glow != glow;
}

/// 2-second selection window shared by all three faces (§7 client throttle).
/// Geometry and colors still come from tokens only — §2.1.4.
class EmojiPicker extends StatefulWidget {
  const EmojiPicker({
    super.key,
    required this.onSelected,
    this.maxFaceSize = 180,
    this.throttle = const Duration(seconds: 2),
    this.autofocusFirst = false,
    this.faceAnimationBackend,
  })  : assert(maxFaceSize > 0, 'maxFaceSize must be positive'),
        assert(throttle > Duration.zero, 'throttle must be positive');

  /// Fired once per accepted activation. Activations inside [throttle]
  /// never reach this callback — the submit path stays single-shot.
  final ValueChanged<EmojiChoice> onSelected;

  /// Upper bound for one face. Phones scale down from here; a 4K TV can
  /// raise it without touching the painter.
  final double maxFaceSize;

  /// §7 client throttle: a second activation inside this window is dropped.
  final Duration throttle;

  /// Requests focus on the first (happy) face after the first frame.
  /// Used by the TV pass; I24 decides the phone default.
  final bool autofocusFirst;

  /// Optional animated backend for the faces (I30 Rive seam). When null the
  /// proven `CustomPainter` path is used unchanged.
  final SketchAnimationBackend? faceAnimationBackend;

  @override
  State<EmojiPicker> createState() => _EmojiPickerState();
}

class _EmojiPickerState extends State<EmojiPicker> {
  Timer? _throttleTimer;

  bool get _isThrottled => _throttleTimer?.isActive ?? false;

  void _select(EmojiChoice choice) {
    if (_isThrottled) {
      return;
    }
    // The inactive timer itself is the release signal — no wall clock.
    _throttleTimer = Timer(widget.throttle, _endThrottle);
    widget.onSelected(choice);
  }

  void _endThrottle() {
    // Intentionally empty: timer inactivity is the release signal.
  }

  @override
  void dispose() {
    _throttleTimer?.cancel();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return LayoutBuilder(
      builder: (context, constraints) {
        // Each face is a third of the row minus breathing room. The cap
        // keeps 10-foot sizing from turning into a giant on wide screens.
        final raw = (constraints.maxWidth - AppSpacing.lg * 2) / 3;
        final faceSize = raw.isFinite
            ? raw.clamp(0.0, widget.maxFaceSize).toDouble()
            : widget.maxFaceSize;

        return Row(
          mainAxisAlignment: MainAxisAlignment.spaceEvenly,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            for (final choice in EmojiChoice.values)
              EmojiFace(
                choice: choice,
                size: faceSize,
                autofocus: widget.autofocusFirst &&
                    choice == EmojiChoice.values.first,
                animationBackend: widget.faceAnimationBackend,
                onActivate: () => _select(choice),
              ),
          ],
        );
      },
    );
  }
}

/// Paints one face as three or four wobbly ink strokes on paper.
///
/// Deliberately cheap: a handful of paths per face, one `CustomPaint` each,
/// no shaders or save layers — safe for a low-end TV GPU (§2.3.3).
class _EmojiFacePainter extends CustomPainter {
  const _EmojiFacePainter({
    required this.choice,
    required this.ink,
    required this.accent,
    required this.blush,
  });

  final EmojiChoice choice;
  final Color ink;
  final Color accent;
  final Color blush;

  static const double _washOpacity = 0.16;
  static const double _blushOpacity = 0.5;

  /// Deterministic wobble phase per face so repaints never shimmer.
  int get _seed => switch (choice) {
        EmojiChoice.satisfied => 3,
        EmojiChoice.neutral => 11,
        EmojiChoice.dissatisfied => 19,
      };

  @override
  void paint(Canvas canvas, Size size) {
    final side = size.shortestSide;
    final center = Offset(size.width / 2, size.height / 2);

    // A soft accent swipe behind the face, like a marker highlight.
    final wash = Paint()
      ..color = accent.withValues(alpha: _washOpacity)
      ..style = PaintingStyle.fill;
    canvas.drawPath(
      _blob(
        center.translate(side * 0.01, side * 0.015),
        side * 0.46,
        seed: _seed + 5,
      ),
      wash,
    );

    final stroke = Paint()
      ..color = ink
      ..style = PaintingStyle.stroke
      ..strokeWidth = side * 0.018
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;

    // A faint second pass, offset by a hair — the "drawn twice" pencil look.
    final ghost = Paint()
      ..color = ink.withValues(alpha: 0.28)
      ..style = PaintingStyle.stroke
      ..strokeWidth = side * 0.012
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;
    canvas.drawPath(
      _blob(
        center.translate(-side * 0.007, side * 0.006),
        side * 0.405,
        seed: _seed + 1,
      ),
      ghost,
    );

    canvas.drawPath(_blob(center, side * 0.41, seed: _seed), stroke);
    _paintFeatures(canvas, side, center, stroke);
  }

  void _paintFeatures(Canvas canvas, double side, Offset center, Paint stroke) {
    final eyeY = center.dy - side * 0.07;
    final leftEye = Offset(center.dx - side * 0.16, eyeY);
    final rightEye = Offset(center.dx + side * 0.16, eyeY);

    if (choice == EmojiChoice.satisfied) {
      _drawClosedEye(canvas, leftEye, side, stroke);
      _drawClosedEye(canvas, rightEye, side, stroke);
      _drawCheeks(canvas, center, side);
    } else {
      _drawDotEye(canvas, leftEye, side);
      _drawDotEye(canvas, rightEye, side);
    }

    _drawMouth(canvas, center, side, stroke);
  }

  /// A happy "^" eye: an upward-bowed arc.
  void _drawClosedEye(Canvas canvas, Offset eye, double side, Paint stroke) {
    final width = side * 0.13;
    final path = Path()
      ..moveTo(eye.dx - width / 2, eye.dy + width * 0.12)
      ..quadraticBezierTo(
        eye.dx,
        eye.dy - width * 0.45,
        eye.dx + width / 2,
        eye.dy + width * 0.12,
      );
    canvas.drawPath(path, stroke);
  }

  void _drawDotEye(Canvas canvas, Offset eye, double side) {
    final dot = Paint()
      ..color = ink
      ..style = PaintingStyle.fill;
    canvas.drawCircle(eye, side * 0.027, dot);
  }

  void _drawCheeks(Canvas canvas, Offset center, double side) {
    final cheek = Paint()
      ..color = blush.withValues(alpha: _blushOpacity)
      ..style = PaintingStyle.fill;
    final y = center.dy + side * 0.1;
    canvas.drawCircle(
      Offset(center.dx - side * 0.27, y),
      side * 0.05,
      cheek,
    );
    canvas.drawCircle(
      Offset(center.dx + side * 0.27, y),
      side * 0.05,
      cheek,
    );
  }

  void _drawMouth(Canvas canvas, Offset center, double side, Paint stroke) {
    final halfWidth = side * 0.15;
    final mouthY = center.dy + side * 0.12;
    final path = Path();

    switch (choice) {
      case EmojiChoice.satisfied:
        // A big "U" smile.
        path
          ..moveTo(center.dx - halfWidth, mouthY)
          ..quadraticBezierTo(
            center.dx,
            mouthY + side * 0.16,
            center.dx + halfWidth,
            mouthY,
          );
      case EmojiChoice.neutral:
        // A small wavy line — not a frown, not a smile.
        path
          ..moveTo(center.dx - halfWidth, mouthY + side * 0.02)
          ..quadraticBezierTo(
            center.dx - halfWidth / 2,
            mouthY - side * 0.01,
            center.dx,
            mouthY + side * 0.02,
          )
          ..quadraticBezierTo(
            center.dx + halfWidth / 2,
            mouthY + side * 0.05,
            center.dx + halfWidth,
            mouthY + side * 0.02,
          );
      case EmojiChoice.dissatisfied:
        // An inverted "U" frown: corners down, centre up.
        path
          ..moveTo(center.dx - halfWidth, mouthY + side * 0.07)
          ..quadraticBezierTo(
            center.dx,
            mouthY - side * 0.1,
            center.dx + halfWidth,
            mouthY + side * 0.07,
          );
    }

    canvas.drawPath(path, stroke);
  }

  /// A closed but slightly irregular circle, sampled with two low-frequency
  /// sine waves — hand-drawn without randomness.
  Path _blob(Offset center, double radius, {required int seed}) {
    const steps = 48;
    final path = Path();

    for (var i = 0; i <= steps; i++) {
      final angle = 2 * math.pi * i / steps;
      final wobble = 1 +
          0.018 * math.sin(3 * angle + seed) +
          0.011 * math.sin(5 * angle - seed * 1.7);
      final point = center +
          Offset(
            math.cos(angle) * radius * wobble,
            math.sin(angle) * radius * wobble,
          );

      if (i == 0) {
        path.moveTo(point.dx, point.dy);
      } else {
        path.lineTo(point.dx, point.dy);
      }
    }

    return path..close();
  }

  @override
  bool shouldRepaint(covariant _EmojiFacePainter oldDelegate) =>
      oldDelegate.choice != choice ||
      oldDelegate.ink != ink ||
      oldDelegate.accent != accent ||
      oldDelegate.blush != blush;
}

/// Faint dot-grid "paper texture" (§5.1).
///
/// One `drawPoints` call instead of ~14k `drawCircle` calls on a 4K TV —
/// the difference between a texture and a jank source (§2.3.3).
class _DotGridPainter extends CustomPainter {
  const _DotGridPainter();

  static const double _dotOpacity = 0.06;
  static const double _dotRadius = 1.2;

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = AppColors.inkSecondary.withValues(alpha: _dotOpacity)
      ..strokeWidth = _dotRadius * 2
      ..strokeCap = StrokeCap.round;

    final points = <Offset>[
      for (var x = AppSpacing.xl / 2; x < size.width; x += AppSpacing.xl)
        for (var y = AppSpacing.xl / 2; y < size.height; y += AppSpacing.xl)
          Offset(x, y),
    ];

    canvas.drawPoints(PointMode.points, points, paint);
  }

  @override
  bool shouldRepaint(covariant _DotGridPainter oldDelegate) => false;
}

/// Wobbly strip of masking tape holding the polaroid up.
class _TapePainter extends CustomPainter {
  const _TapePainter({required this.tape, required this.edge});

  final Color tape;
  final Color edge;

  @override
  void paint(Canvas canvas, Size size) {
    final strip = Path()
      ..moveTo(3, 4)
      ..lineTo(size.width - 4, 1)
      ..lineTo(size.width - 1, size.height - 3)
      ..lineTo(2, size.height - 5)
      ..close();

    canvas.drawPath(strip, Paint()..color = tape..style = PaintingStyle.fill);
    canvas.drawPath(
      strip,
      Paint()
        ..color = edge.withValues(alpha: 0.3)
        ..style = PaintingStyle.stroke
        ..strokeWidth = 1.2
        ..strokeJoin = StrokeJoin.round,
    );
  }

  @override
  bool shouldRepaint(covariant _TapePainter oldDelegate) =>
      oldDelegate.tape != tape || oldDelegate.edge != edge;
}

/// Hand-drawn page dots: wobbly ellipses, the active one filled with the
/// warm accent. No animation controller — cheap on a low-end TV (§2.3.3).
class _PageDotsPainter extends CustomPainter {
  const _PageDotsPainter({
    required this.count,
    required this.active,
    required this.ink,
    required this.accent,
  });

  final int count;
  final int active;
  final Color ink;
  final Color accent;

  @override
  void paint(Canvas canvas, Size size) {
    final spacing = size.width / (count + 1);

    for (var i = 0; i < count; i++) {
      final isActive = i == active;
      final center = Offset(
        spacing * (i + 1),
        size.height / 2 + (i.isEven ? 1 : -1),
      );
      final radius = isActive ? 6.0 : 4.0;
      final path = Path();
      const steps = 16;

      for (var step = 0; step <= steps; step++) {
        final angle = 2 * math.pi * step / steps;
        final wobble = 1 + 0.08 * math.sin(3 * angle + i);
        final point = center +
            Offset(
              math.cos(angle) * radius * wobble,
              math.sin(angle) * radius * wobble,
            );
        if (step == 0) {
          path.moveTo(point.dx, point.dy);
        } else {
          path.lineTo(point.dx, point.dy);
        }
      }

      canvas.drawPath(
        path..close(),
        Paint()
          ..color = isActive ? accent : ink.withValues(alpha: 0.4)
          ..style = isActive ? PaintingStyle.fill : PaintingStyle.stroke
          ..strokeWidth = 2,
      );
    }
  }

  @override
  bool shouldRepaint(covariant _PageDotsPainter oldDelegate) =>
      oldDelegate.count != count ||
      oldDelegate.active != active ||
      oldDelegate.ink != ink ||
      oldDelegate.accent != accent;
}

/// Draws a wobbly five-point star with a faint second pencil pass.
class _StarPainter extends CustomPainter {
  const _StarPainter({
    required this.progress,
    required this.ink,
    required this.gold,
  });

  final double progress;
  final Color ink;
  final Color gold;

  @override
  void paint(Canvas canvas, Size size) {
    final path = _starPath(size);

    // Every star is visible, selected or not — only the gold fill draws in.
    canvas.save();
    canvas.translate(1, 1);
    canvas.drawPath(
      path,
      Paint()
        ..color = ink.withValues(alpha: 0.3)
        ..style = PaintingStyle.stroke
        ..strokeWidth = 1.4
        ..strokeJoin = StrokeJoin.round,
    );
    canvas.restore();

    if (progress > 0) {
      canvas.drawPath(
        path,
        Paint()
          ..color = gold.withValues(alpha: 0.55 * progress)
          ..style = PaintingStyle.fill,
      );

      // A fat pencil stroke walks around the star, filling it in.
      final fill = Paint()
        ..color = gold
        ..style = PaintingStyle.stroke
        ..strokeWidth = size.shortestSide * 0.16
        ..strokeCap = StrokeCap.round
        ..strokeJoin = StrokeJoin.round;
      for (final metric in path.computeMetrics()) {
        canvas.drawPath(
          metric.extractPath(0, metric.length * progress),
          fill,
        );
      }
    }

    canvas.drawPath(
      path,
      Paint()
        ..color = ink
        ..style = PaintingStyle.stroke
        ..strokeWidth = 2.4
        ..strokeJoin = StrokeJoin.round,
    );
  }

  Path _starPath(Size size) {
    final center = Offset(size.width / 2, size.height / 2);
    final outer = size.shortestSide * 0.44;
    final inner = outer * 0.45;
    final path = Path();

    for (var i = 0; i < 10; i++) {
      final angle = -math.pi / 2 + i * math.pi / 5;
      final radius = i.isEven ? outer : inner;
      final wobble = 1 + 0.04 * math.sin(i * 2.7);
      final point = center +
          Offset(
            math.cos(angle) * radius * wobble,
            math.sin(angle) * radius * wobble,
          );
      if (i == 0) {
        path.moveTo(point.dx, point.dy);
      } else {
        path.lineTo(point.dx, point.dy);
      }
    }

    return path..close();
  }

  @override
  bool shouldRepaint(covariant _StarPainter oldDelegate) =>
      oldDelegate.progress != progress ||
      oldDelegate.ink != ink ||
      oldDelegate.gold != gold;
}

/// An injectable animated-visual backend (I30, §5.1).
///
/// Every backend must be able to render the supplied [fallback] painter —
/// a kiosk must never show a broken or empty animation asset.
abstract interface class SketchAnimationBackend {
  Widget build(
    BuildContext context, {
    required Widget fallback,
    required bool active,
  });
}

/// Pure no-op: renders the painter unchanged. Use in tests and anywhere an
/// animation is not wanted (no loaders, no tickers, no timers).
final class NoOpSketchAnimationBackend implements SketchAnimationBackend {
  const NoOpSketchAnimationBackend();

  @override
  Widget build(
    BuildContext context, {
    required Widget fallback,
    required bool active,
  }) =>
      fallback;
}

/// The production default while real `.riv` art is unavailable: the
/// hand-drawn painter itself gets a small animated response to `active`.
final class FallbackSketchAnimationBackend implements SketchAnimationBackend {
  const FallbackSketchAnimationBackend();

  @override
  Widget build(
    BuildContext context, {
    required Widget fallback,
    required bool active,
  }) {
    return AnimatedScale(
      scale: active ? 1.06 : 1.0,
      duration: const Duration(milliseconds: 250),
      curve: Curves.easeOutBack,
      child: AnimatedOpacity(
        opacity: active ? 1 : 0.85,
        duration: const Duration(milliseconds: 250),
        child: fallback,
      ),
    );
  }
}

/// A sketch visual with two states and a guaranteed `CustomPainter` fallback.
///
/// The backend is injected so tests can be hermetic and a missing or broken
/// `.riv` asset can never leave a blank kiosk screen.
class SketchAnimation extends StatelessWidget {
  const SketchAnimation({
    super.key,
    required this.painter,
    this.backend = const FallbackSketchAnimationBackend(),
    this.active = false,
    this.fallbackKey,
    this.size,
  });

  /// The always-available hand-drawn visual.
  final CustomPainter painter;

  /// How the visual is animated; defaults to the painter fallback.
  final SketchAnimationBackend backend;

  /// Current state (e.g. a selected face or a waving character).
  final bool active;

  /// Key preserved on the fallback painter for existing component tests.
  final Key? fallbackKey;

  /// Optional square size constraint.
  final double? size;

  @override
  Widget build(BuildContext context) {
    Widget fallback = CustomPaint(key: fallbackKey, painter: painter);
    if (size != null) {
      fallback = SizedBox.square(dimension: size!, child: fallback);
    }

    try {
      return backend.build(context, fallback: fallback, active: active);
    } catch (_) {
      // A hostile/buggy backend cannot break the kiosk.
      return fallback;
    }
  }
}

import 'dart:math' as math;

import 'package:flutter/material.dart';

/// Draws the sketch focus frame as two jittered pencil passes that "write"
/// themselves in as [progress] runs from 0 to 1 (§5.1, §2.3.2).
///
/// Deliberately in-house: `rough_notation`'s autoplay timer calls
/// `AnimationController.forward()` after dispose when focus moves quickly,
/// which is a real race on a D-pad. This painter owns no timers — its owner
/// supplies the [Animation] (normally an `AnimationController`).
///
/// Shared by the emoji faces (I19) and the rating prompt (I23) so every
/// focusable kiosk control has the same unmistakable focus affordance.
class SketchFocusFramePainter extends CustomPainter {
  SketchFocusFramePainter({
    required this.progress,
    required this.color,
    required this.strokeWidth,
  }) : super(repaint: progress);

  final Animation<double> progress;
  final Color color;
  final double strokeWidth;

  @override
  void paint(Canvas canvas, Size size) {
    final t = progress.value;
    if (t <= 0) {
      return;
    }

    final paint = Paint()
      ..color = color
      ..style = PaintingStyle.stroke
      ..strokeWidth = strokeWidth
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;

    // Two passes, different jitter — a box drawn twice by hand.
    for (final seed in const [7, 23]) {
      final path = _jitteredRect(size, seed);
      for (final metric in path.computeMetrics()) {
        canvas.drawPath(metric.extractPath(0, metric.length * t), paint);
      }
    }
  }

  Path _jitteredRect(Size size, int seed) {
    final random = math.Random(seed);
    Offset corner(double x, double y) => Offset(
          x + random.nextDouble() * 5 - 2.5,
          y + random.nextDouble() * 5 - 2.5,
        );

    final topLeft = corner(0, 0);
    final topRight = corner(size.width, 0);
    final bottomRight = corner(size.width, size.height);
    final bottomLeft = corner(0, size.height);

    return Path()
      ..moveTo(topLeft.dx, topLeft.dy)
      ..lineTo(topRight.dx, topRight.dy)
      ..lineTo(bottomRight.dx, bottomRight.dy)
      ..lineTo(bottomLeft.dx, bottomLeft.dy)
      ..close();
  }

  @override
  bool shouldRepaint(covariant SketchFocusFramePainter oldDelegate) =>
      oldDelegate.color != color || oldDelegate.strokeWidth != strokeWidth;
}

/// Wobbly speech bubble with a tail pointing down at the character.
class _SpeechBubblePainter extends CustomPainter {
  const _SpeechBubblePainter({required this.ink, required this.paper});

  final Color ink;
  final Color paper;

  @override
  void paint(Canvas canvas, Size size) {
    final bubble = _bubblePath(size);

    canvas.drawPath(
      bubble,
      Paint()..color = paper..style = PaintingStyle.fill,
    );

    // A faint offset second pass keeps the outline hand-drawn.
    canvas.save();
    canvas.translate(1.5, 1);
    canvas.drawPath(
      bubble,
      Paint()
        ..color = ink.withValues(alpha: 0.28)
        ..style = PaintingStyle.stroke
        ..strokeWidth = 1.6
        ..strokeCap = StrokeCap.round
        ..strokeJoin = StrokeJoin.round,
    );
    canvas.restore();

    canvas.drawPath(
      bubble,
      Paint()
        ..color = ink
        ..style = PaintingStyle.stroke
        ..strokeWidth = 2.4
        ..strokeCap = StrokeCap.round
        ..strokeJoin = StrokeJoin.round,
    );
  }

  Path _bubblePath(Size size) {
    final w = size.width;
    final bottom = size.height - 12; // leave room for the tail
    return Path()
      ..moveTo(20, 5)
      ..lineTo(w - 22, 7)
      ..quadraticBezierTo(w - 3, 8, w - 5, 24)
      ..lineTo(w - 7, bottom - 20)
      ..quadraticBezierTo(w - 9, bottom - 2, w - 26, bottom - 4)
      ..lineTo(w * 0.58, bottom - 4)
      ..lineTo(w * 0.52, bottom + 10)
      ..lineTo(w * 0.46, bottom - 4)
      ..lineTo(24, bottom - 4)
      ..quadraticBezierTo(4, bottom - 6, 5, bottom - 24)
      ..lineTo(3, 24)
      ..quadraticBezierTo(4, 6, 20, 5)
      ..close();
  }

  @override
  bool shouldRepaint(covariant _SpeechBubblePainter oldDelegate) =>
      oldDelegate.ink != ink || oldDelegate.paper != paper;
}

/// Simple waving stick character — ink strokes, paper fill, warm cheeks.
class _CharacterPainter extends CustomPainter {
  const _CharacterPainter({
    required this.ink,
    required this.paper,
    required this.blush,
    required this.wave,
  });

  final Color ink;
  final Color paper;
  final Color blush;
  final Color wave;

  @override
  void paint(Canvas canvas, Size size) {
    final centerX = size.width / 2;

    final stroke = Paint()
      ..color = ink
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2.6
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;

    // Head.
    final head = Offset(centerX, size.height * 0.27);
    final radius = size.width * 0.18;
    canvas.drawCircle(
      head,
      radius,
      Paint()..color = paper..style = PaintingStyle.fill,
    );
    canvas.drawCircle(head, radius, stroke);

    // Eyes + smile.
    final eyeY = head.dy - radius * 0.1;
    canvas.drawCircle(
      head.translate(-radius * 0.36, 0),
      2.2,
      Paint()..color = ink..style = PaintingStyle.fill,
    );
    canvas.drawCircle(
      head.translate(radius * 0.36, 0),
      2.2,
      Paint()..color = ink..style = PaintingStyle.fill,
    );
    canvas.drawPath(
      Path()
        ..moveTo(head.dx - radius * 0.4, eyeY + radius * 0.35)
        ..quadraticBezierTo(
          head.dx,
          eyeY + radius * 0.85,
          head.dx + radius * 0.4,
          eyeY + radius * 0.35,
        ),
      stroke,
    );

    // Cheeks.
    final cheek = Paint()
      ..color = blush.withValues(alpha: 0.5)
      ..style = PaintingStyle.fill;
    canvas.drawCircle(
      head.translate(-radius * 0.72, radius * 0.35),
      3.4,
      cheek,
    );
    canvas.drawCircle(
      head.translate(radius * 0.72, radius * 0.35),
      3.4,
      cheek,
    );

    // Body, legs and the raised waving arm.
    final shoulder = Offset(centerX, size.height * 0.52);
    canvas.drawPath(
      Path()
        ..moveTo(shoulder.dx, shoulder.dy)
        ..quadraticBezierTo(
          shoulder.dx + 3,
          size.height * 0.66,
          shoulder.dx,
          size.height * 0.76,
        ),
      stroke,
    );
    canvas.drawLine(
      shoulder,
      Offset(centerX - size.width * 0.22, size.height * 0.66),
      stroke,
    );
    canvas.drawLine(
      shoulder,
      Offset(centerX + size.width * 0.26, size.height * 0.38),
      stroke,
    );
    canvas.drawLine(
      Offset(centerX, size.height * 0.76),
      Offset(centerX - size.width * 0.14, size.height * 0.94),
      stroke,
    );
    canvas.drawLine(
      Offset(centerX, size.height * 0.76),
      Offset(centerX + size.width * 0.14, size.height * 0.94),
      stroke,
    );

    // Little motion strokes around the waving hand.
    final motion = Paint()
      ..color = wave
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2
      ..strokeCap = StrokeCap.round;
    final handX = centerX + size.width * 0.26;
    final handY = size.height * 0.38;
    for (final offset in const [-7.0, 0.0, 7.0]) {
      canvas.drawLine(
        Offset(handX + offset, handY - 6),
        Offset(handX + offset + 2, handY - 12),
        motion,
      );
    }
  }

  @override
  bool shouldRepaint(covariant _CharacterPainter oldDelegate) =>
      oldDelegate.ink != ink ||
      oldDelegate.paper != paper ||
      oldDelegate.blush != blush ||
      oldDelegate.wave != wave;
}
